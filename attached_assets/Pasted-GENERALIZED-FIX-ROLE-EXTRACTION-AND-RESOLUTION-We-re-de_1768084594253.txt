```python
def resolve_role(self, role_query: str, tenant_id: str) -> Optional[str]:
    """
    Resolve a role to a person name.
    
    Handles: CEO, Chief Executive Officer, chief executive, ceo, etc.
    """
    # Normalize the role query
    normalized = self._normalize_role(role_query)  # "ceo" → "CEO"
    
    # Stage 1: Exact relationship lookup
    result = self._lookup_holds_position(normalized, tenant_id)
    if result:
        return result
    
    # Stage 2: Fuzzy relationship lookup (ILIKE)
    result = self._fuzzy_lookup_holds_position(role_query, tenant_id)
    if result:
        return result
    
    # Stage 3: Find role entity, then find holder
    role_entity = self.entity_resolver.resolve(role_query, tenant_id)
    if role_entity:
        result = self._find_position_holder(role_entity.id, tenant_id)
        if result:
            return result
    
    return None

def _normalize_role(self, role: str) -> str:
    """Normalize role variations to standard form."""
    role_map = {
        'chief executive officer': 'CEO',
        'chief technology officer': 'CTO', 
        'chief financial officer': 'CFO',
        'chief operating officer': 'COO',
        'chief data officer': 'CDO',
        'ceo': 'CEO',
        'cto': 'CTO',
        'cfo': 'CFO',
        'coo': 'COO',
        'cdo': 'CDO',
    }
    return role_map.get(role.lower().strip(), role.upper())
```
