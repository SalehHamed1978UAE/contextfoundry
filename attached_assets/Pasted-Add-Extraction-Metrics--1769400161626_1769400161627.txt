  @app.route('/api/extraction/metrics/<vault_id>')                                                       
  def get_extraction_metrics(vault_id):                                                                  
      """Get entity and relationship counts by extraction level."""                                      
      with get_db_session() as session:                                                                  
          # Get counts by extraction level                                                               
          stats = session.execute(text("""                                                               
              SELECT                                                                                     
                  d.extraction_level,                                                                    
                  COUNT(DISTINCT e.id) as entity_count,                                                  
                  COUNT(DISTINCT r.id) as relationship_count                                             
              FROM platform.documents d                                                                  
              LEFT JOIN entities e ON e.document_id = d.id                                               
              LEFT JOIN relationships r ON r.document_id = d.id                                          
              WHERE d.tenant_id = :vault_id                                                              
              GROUP BY d.extraction_level                                                                
          """), {"vault_id": vault_id}).fetchall()                                                       
                                                                                                         
          result = {                                                                                     
              "single": {"entities": 0, "relationships": 0},                                             
              "multi": {"entities": 0, "relationships": 0}                                               
          }                                                                                              
                                                                                                         
          for row in stats:                                                                              
              level = row.extraction_level or "single"                                                   
              if level in result:                                                                        
                  result[level]["entities"] = row.entity_count                                           
                  result[level]["relationships"] = row.relationship_count                                
                                                                                                         
          # Calculate improvement                                                                        
          if result["single"]["entities"] > 0:                                                           
              result["improvement"] = {                                                                  
                  "entities_pct": round((result["multi"]["entities"] / result["single"]["entities"] - 1) 
  * 100, 1),                                                                                             
                  "relationships_pct": round((result["multi"]["relationships"] /                         
  result["single"]["relationships"] - 1) * 100, 1)                                                       
              }                                                                                          
                                                                                                         
          return jsonify(result)                                                                         
                                                                                                         
  2. For Phase 1 (JSON files, before ingestion)                                                          
                                                                                                         
  Count from extraction output files:                                                                    
                                                                                                         
  @app.route('/api/extraction/metrics/phase1/<vault_id>')                                                
  def get_phase1_metrics(vault_id):                                                                      
      """Get entity/relationship counts from extraction JSON files."""                                   
      import json                                                                                        
                                                                                                         
      output_dir = Path(f"extraction_output/{vault_id}")                                                 
                                                                                                         
      gpt_entities = 0                                                                                   
      gpt_relationships = 0                                                                              
      claude_entities = 0                                                                                
      claude_relationships = 0                                                                           
                                                                                                         
      # Count GPT extractions                                                                            
      gpt_dir = output_dir / "gpt_4o_mini"                                                               
      if gpt_dir.exists():                                                                               
          for f in gpt_dir.glob("*.json"):                                                               
              data = json.loads(f.read_text())                                                           
              gpt_entities += len(data.get("entities", []))                                              
              gpt_relationships += len(data.get("relationships", []))                                    
                                                                                                         
      # Count Claude extractions                                                                         
      claude_dir = output_dir / "claude_sonnet"                                                          
      if claude_dir.exists():                                                                            
          for f in claude_dir.glob("*.json"):                                                            
              data = json.loads(f.read_text())                                                           
              claude_entities += len(data.get("entities", []))                                           
              claude_relationships += len(data.get("relationships", []))                                 
                                                                                                         
      return jsonify({                                                                                   
          "gpt_4o_mini": {                                                                               
              "entities": gpt_entities,                                                                  
              "relationships": gpt_relationships                                                         
          },                                                                                             
          "claude_sonnet": {                                                                             
              "entities": claude_entities,                                                               
              "relationships": claude_relationships                                                      
          },                                                                                             
          "combined_potential": {                                                                        
              "entities": max(gpt_entities, claude_entities),                                            
              "relationships": max(gpt_relationships, claude_relationships)                              
          }                                                                                              
      })
