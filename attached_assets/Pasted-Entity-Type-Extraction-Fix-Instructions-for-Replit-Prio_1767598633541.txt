VALID_ENTITY_TYPES = {"SERVICE", "DATABASE", "TEAM", "PERSON", "INCIDENT", "SYSTEM", "COMPONENT"}

TYPE_MAPPING = {
    # Map generic types to schema types
    "ORGANIZATION": "TEAM",      # Internal teams
    "PROCESS": "SERVICE",        # Software processes
    "EVENT": "INCIDENT",         # Operational events
    "APPLICATION": "SERVICE",    # Apps are services
    "API": "SERVICE",            # APIs are services
    "PLATFORM": "SYSTEM",        # Platforms are systems
    "GROUP": "TEAM",             # Groups are teams
}

def validate_entity_type(extracted_type: str, entity_name: str) -> str:
    """Validate and correct entity type."""
    
    extracted_type = extracted_type.upper()
    
    # Already valid
    if extracted_type in VALID_ENTITY_TYPES:
        return extracted_type
    
    # Map to valid type
    if extracted_type in TYPE_MAPPING:
        return TYPE_MAPPING[extracted_type]
    
    # Heuristic fallbacks based on name patterns
    name_lower = entity_name.lower()
    
    if any(x in name_lower for x in ['service', 'gateway', 'api', 'portal']):
        return "SERVICE"
    if any(x in name_lower for x in ['database', 'db', 'store', 'warehouse']):
        return "DATABASE"
    if any(x in name_lower for x in ['team', 'group', 'department']):
        return "TEAM"
    if any(x in name_lower for x in ['inc-', 'incident', 'outage']):
        return "INCIDENT"
    
    # Default fallback
    return "SERVICE"  # Most common type
