    try:
        # First, trigger extraction for each document
        self.log("         Triggering extraction...")
        
        for doc_id in self.document_ids:
            # Call the extraction function directly
            # (or use the API endpoint if authentication is set up)
            from services.extraction import trigger_extraction
            trigger_extraction(doc_id, self.tenant_id)
            self.log(f"         - Document {doc_id}: extraction triggered")
        
        # Now poll for completion
        self.log(f"         Polling every {EXTRACTION_POLL_INTERVAL_SECONDS}s (max {MAX_EXTRACTION_WAIT_SECONDS}s)...")
        
        session = get_session(use_rls_role=False)
        elapsed = 0
        
        while elapsed < MAX_EXTRACTION_WAIT_SECONDS:
            # Check counts
            result = session.execute(
                text("""
                    SELECT 
                        (SELECT COUNT(*) FROM document_chunks WHERE tenant_id = :tid) as chunks,
                        (SELECT COUNT(*) FROM entities WHERE tenant_id = :tid) as entities,
                        (SELECT COUNT(*) FROM relationships WHERE tenant_id = :tid) as rels
                """),
                {"tid": self.tenant_id}
            ).fetchone()
            
            chunks, entities, rels = result
            
            if elapsed % 30 == 0 or (chunks > 0 and entities > 0 and rels > 0):
                self.log(f"         [{elapsed}s] chunks: {chunks}, entities: {entities}, relationships: {rels}")
            
            if chunks > 0 and entities > 0 and rels > 0:
                self.log(f"         Result: OK (completed in {elapsed}s)")
                self.log(f"         - Chunks: {chunks}")
                self.log(f"         - Entities: {entities}")
                self.log(f"         - Relationships: {rels}")
                self.extraction_counts = {"chunks": chunks, "entities": entities, "relationships": rels}
                return True
            
            time.sleep(EXTRACTION_POLL_INTERVAL_SECONDS)
            elapsed += EXTRACTION_POLL_INTERVAL_SECONDS
        
        # Timeout
        self.fail("Step 3", f"Extraction timeout after {MAX_EXTRACTION_WAIT_SECONDS}s. chunks={chunks}, entities={entities}, rels={rels}")
        return False
        
    except Exception as e:
        self.fail("Step 3", f"Extraction failed: {e}")
        return False

def step4_verify_extraction(self) -> bool:
    self.log("Step 4: Verify Extraction")
    
    session = get_session(use_rls_role=False)
    missing_entities = []
    missing_relationships = []
    
    try:
        # Check required entities
        self.log("         Checking required entities...")
        
        required_entities = [
            ("Sarah Chen", "PERSON"),
            ("Marcus Williams", "PERSON"),
            ("TechVentures", "ORGANIZATION"),
            ("CloudMatrix", "ORGANIZATION"),
        ]
        
        for name, entity_type in required_entities:
            result = session.execute(
                text("""
                    SELECT id FROM entities 
                    WHERE tenant_id = :tid 
                    AND name ILIKE :name
                    AND entity_type = :etype
                """),
                {"tid": self.tenant_id, "name": f"%{name}%", "etype": entity_type}
            ).fetchone()
            
            if result:
                self.log(f"         - {name} ({entity_type}): FOUND")
            else:
                self.log(f"         - {name} ({entity_type}): MISSING")
                missing_entities.append(f"{name} ({entity_type})")
        
        # Check required relationships
        self.log("         Checking required relationships...")
        
        # Check Sarah Chen has CEO role
        result = session.execute(
            text("""
                SELECT e.id FROM entities e
                LEFT JOIN relationships r ON e.id = r.source_entity_id
                LEFT JOIN entities target ON r.target_entity_id = target.id
                WHERE e.tenant_id = :tid
                AND e.name ILIKE '%Sarah Chen%'
                AND (
                    e.properties->>'position' ILIKE '%CEO%'
                    OR e.properties->>'position' ILIKE '%Chief Executive%'
                    OR target.name ILIKE '%CEO%'
                    OR target.name ILIKE '%Chief Executive%'
                )
            """),
            {"tid": self.tenant_id}
        ).fetchone()
        
        if result:
            self.log("         - Sarah Chen → CEO: FOUND")
        else:
            self.log("         - Sarah Chen → CEO: MISSING")
            missing_relationships.append("Sarah Chen → CEO")
        
        # Check Marcus Williams has CTO role
        result = session.execute(
            text("""
                SELECT e.id FROM entities e

                WHERE e.tenant_id = :tid
                AND e.name ILIKE '%Marcus Williams%'
                AND (
                    e.properties->>'position' ILIKE '%CTO%'
                    OR e.properties->>'position' ILIKE '%Chief Technology%'

            {"tid": self.tenant_id}
        ).fetchone()

            self.log("         - Marcus Williams → CTO: MISSING")
            missing_relationships.append("Marcus Williams → CTO")

def step5_run_queries(self) -> bool:
    self.log("Step 5: Run Queries")
    
    from tool_agent import ToolAgent
    
    test_queries = [
        {
            "query": "Who is the CEO?",
            "expected_contains": ["Sarah Chen"],
            "min_confidence": 0.70,
        },
        {
            "query": "Who is the CTO?",
            "expected_contains": ["Marcus Williams"],
            "min_confidence": 0.70,
        },
        {
            "query": "What is Sarah Chen's compensation?",
            "expected_contains": ["850,000", "850000", "$850,000"],
            "match_any": True,
            "min_confidence": 0.70,
        },
    ]
    
    session = get_session(use_rls_role=True)
    # Set RLS context
    session.execute(text(f"SET app.current_tenant_id = '{self.tenant_id}'"))
    
    agent = ToolAgent(session, self.tenant_id)
    
    passed = 0
    failed = 0
    failures = []
    
    try:
        for i, test in enumerate(test_queries, 1):
            self.log(f"")
            self.log(f"         Query {i}: \"{test['query']}\"")
            
            result = agent.query(test["query"], vault_context=self.vault_name)
            
            answer = result.get("answer", "")
            confidence = result.get("confidence", 0)
            
            self.log(f"         - Answer: {answer[:100]}...")
            self.log(f"         - Confidence: {confidence:.0%}")
            
            # Check expected content
            expected = test["expected_contains"]
            match_any = test.get("match_any", False)
            
            if match_any:
                found = any(exp.lower() in answer.lower() for exp in expected)
            else:
                found = all(exp.lower() in answer.lower() for exp in expected)
            
            # Check confidence
            conf_ok = confidence >= test["min_confidence"]
            
            if found and conf_ok:
                self.log(f"         - Result: PASS")
                passed += 1
            else:
                self.log(f"         - Result: FAIL")
                if not found:
                    self.log(f"           Expected to contain: {expected}")
                if not conf_ok:
                    self.log(f"           Confidence too low: {confidence:.0%} < {test['min_confidence']:.0%}")
                failed += 1
                failures.append(test["query"])
        
        self.log(f"")
        if failed > 0:
            self.fail("Step 5", f"Query tests failed: {failed}/{len(test_queries)}. Failed queries: {failures}")
            return False
        
        self.log(f"         Result: OK ({passed}/{len(test_queries)} passed)")
        return True
        
    except Exception as e:
        self.fail("Step 5", f"Query execution failed: {e}")
        return False
