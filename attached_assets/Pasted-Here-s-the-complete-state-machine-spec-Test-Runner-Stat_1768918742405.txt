pythondef transition_to(run_id: int, new_status: TestRunStatus):
    run = TestRun.query.get(run_id)
    old_status = TestRunStatus(run.status)
    
    if not is_valid_transition(old_status, new_status):
        raise InvalidStateTransition(f"Cannot go from {old_status} to {new_status}")
    
    run.status = new_status.value
    run.updated_at = datetime.utcnow()
    run.heartbeat_at = datetime.utcnow()
    db.session.commit()
    
    log.info(f"Test run {run_id}: {old_status.value} → {new_status.value}")

        transition_to(run_id, TestRunStatus.FAILED)
        raise

# In api.py startup - mark stale tests as interrupted
def startup_cleanup():
    stale_threshold = datetime.utcnow() - timedelta(minutes=5)
    
    stale_runs = TestRun.query.filter(
        TestRun.status.in_(['creating_vault', 'uploading', 'extracting', 'running_qa']),
        TestRun.heartbeat_at < stale_threshold
    ).all()
    
    for run in stale_runs:
        run.status = TestRunStatus.INTERRUPTED.value
        run.error_message = "Interrupted: Server restarted with stale heartbeat"
    
    db.session.commit()
