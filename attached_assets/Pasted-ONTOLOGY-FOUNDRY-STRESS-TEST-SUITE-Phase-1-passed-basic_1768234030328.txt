def test_10_concurrent():
    """Test concurrent writes to same candidate."""
    print("\n" + "=" * 70)
    print("TEST 10: CONCURRENT ACCESS")
    print("=" * 70)
    
    test_type = f"CONCURRENT_{uuid4().hex[:6].upper()}"
    num_threads = 10
    writes_per_thread = 20
    
    def worker(thread_id):
        session = get_session()
        store = CandidateStore(session, TENANT_ID)
        
        for i in range(writes_per_thread):
            try:
                store.add_relationship_candidate(
                    proposed_name=test_type,
                    normalized_name=test_type,
                    source_entity_type="ORGANIZATION",
                    target_entity_type="ORGANIZATION",
                    source_entity_name=f"Thread{thread_id}_Org{i}",
                    target_entity_name=f"Thread{thread_id}_Target{i}",
                    original_text=f"Thread {thread_id} write {i}",
                    document_id=uuid4()
                )
            except Exception as e:
                return (thread_id, False, str(e))
        
        return (thread_id, True, None)
    
    start = time.time()
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=num_threads) as executor:
        futures = [executor.submit(worker, i) for i in range(num_threads)]
        results = [f.result() for f in concurrent.futures.as_completed(futures)]
    
    elapsed = time.time() - start
    
    # Check results
    successful = sum(1 for _, success, _ in results if success)
    print(f"  {successful}/{num_threads} threads completed successfully")
    print(f"  Time: {elapsed:.2f}s")
    
    # Verify final state
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    candidates = store.get_pending_candidates()
    candidate = next((c for c in candidates if c['normalized_name'] == test_type), None)
    
    expected_mentions = num_threads * writes_per_thread
    if candidate:
        print(f"  Final mention count: {candidate['mention_count']}")
        print(f"  Expected: {expected_mentions}")
        
        candidate_id = UUID(candidate['id'])
        extractions = store.get_pending_extractions(candidate_id)
        print(f"  Pending extractions: {len(extractions)}")
    
    # Cleanup
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (
            SELECT id FROM ontology_candidates WHERE normalized_name = '{test_type}'
        )
    """))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{test_type}'"))
    session.commit()
    print(f"  Cleaned up")
    
    # Success if all threads completed and counts match
    success = (successful == num_threads and 
               candidate and 
               candidate['mention_count'] == expected_mentions)
    
    return success

    # Add from same document twice
    store.add_relationship_candidate(
        proposed_name=test_type, normalized_name=test_type,
        source_entity_type="ORG", target_entity_type="ORG",
        source_entity_name="A", target_entity_name="B",
        original_text="A to B", document_id=doc_ids[0]
    )
    store.add_relationship_candidate(
        proposed_name=test_type, normalized_name=test_type,
        source_entity_type="ORG", target_entity_type="ORG",
        source_entity_name="C", target_entity_name="D",
        original_text="C to D", document_id=doc_ids[0]  # Same doc
    )
    
    candidate = next((c for c in store.get_pending_candidates() if c['normalized_name'] == test_type), None)
    if candidate['document_count'] == 1 and candidate['mention_count'] == 2:

    else:
        print(f"    ✗ Expected 1 doc, 2 mentions; got {candidate['document_count']} docs, {candidate['mention_count']} mentions")

    # Add from different documents
    for doc_id in doc_ids[1:]:
        store.add_relationship_candidate(

            source_entity_type="ORG", target_entity_type="ORG",
            source_entity_name="X", target_entity_name="Y",
            original_text="X to Y", document_id=doc_id
        )
    
    candidate = next((c for c in store.get_pending_candidates() if c['normalized_name'] == test_type), None)
    if candidate['document_count'] == 5 and candidate['mention_count'] == 6:
        print(f"    ✓ After 4 more docs: 5 docs, 6 mentions")
    else:
        print(f"    ✗ Expected 5 docs, 6 mentions; got {candidate['document_count']} docs, {candidate['mention_count']} mentions")
        all_passed = False

    print("\n  12b. Example mentions capped at 10:")

            original_text=f"Example {i} text here", document_id=uuid4()

    if len(mentions) <= 10:
        print(f"    ✓ Example mentions capped: {len(mentions)} (max 10)")
    else:
        print(f"    ✗ Too many mentions: {len(mentions)}")
