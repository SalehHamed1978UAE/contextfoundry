```python
CORE_FOUNDATION_TYPES = ['PERSON', 'ORGANIZATION', 'DOCUMENT', 'LOCATION', 'EVENT', 'CONCEPT', 'PROCESS', 'DATE']

def load_domain_schema(domain_preference):
    if domain_preference in DOMAIN_TO_TYPES:
        return DOMAIN_TO_TYPES[domain_preference]
    return CORE_FOUNDATION_TYPES

def load_core_foundation_schema():
    return CORE_FOUNDATION_TYPES
```

Or query from database:

```python
def load_domain_schema(domain_preference):
    # Query ontology.types for this domain's types
    types = db.execute("""
        SELECT type_name FROM ontology.types 
        WHERE status = 'ACTIVE' 
        AND (domain_id = %s OR layer = 1)
        ORDER BY layer, type_name
    """, [domain_preference]).fetchall()
    return [t['type_name'] for t in types]
```
