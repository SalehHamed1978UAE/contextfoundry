  def query(self, question: str, ...):                                                                   
      # STEP 1: LLM interprets what user wants                                                           
      intent = self.query_interpreter.interpret(question)                                                
      logger.info(f"[AGENT] Intent: entity='{intent.entity}', type='{intent.query_type}',                
  target='{intent.target_type}'")                                                                        
                                                                                                         
      # STEP 2: Route based on intent                                                                    
      if intent.query_type == "graph_traversal":                                                         
          # Use KG - don't touch documents                                                               
          kg_result = self.directed_retriever.execute(intent)                                            
          if kg_result.has_data:                                                                         
              return self._synthesize_from_kg(question, kg_result, intent)                               
                                                                                                         
      elif intent.query_type == "property_lookup":                                                       
          # Look up entity properties in KG                                                              
          entity_props = self._lookup_entity_properties(intent.entity)                                   
          if entity_props:                                                                               
              return self._synthesize_from_properties(question, entity_props, intent)                    
                                                                                                         
      # STEP 3: Fall back to document search only if KG doesn't have answer                              
      pipeline_result = self.query_pipeline.process(question, vault_context=vault_context)               
      return self._synthesize_direct_answer(question, pipeline_result, intent)

  def _synthesize_from_kg(self, question, kg_result, intent):                                            
      """Synthesize answer directly from KG data without document search."""                             
                                                                                                         
      # Format KG data                                                                                   
      context = f"Entity: {kg_result.entity_name}\n"                                                     
      if kg_result.properties:                                                                           
          context += f"Properties: {kg_result.properties}\n"                                             
      for rel in kg_result.relationships:                                                                
          context += f"- {rel.source} {rel.type} {rel.target}\n"                                         
                                                                                                         
      prompt = f"""Question: {question}                                                                  
  Query interpretation: {intent.reasoning}                                                               
                                                                                                         
  Knowledge Graph data:                                                                                  
  {context}                                                                                              
                                                                                                         
  Answer based on the structured data above."""                                                          
                                                                                                         
      return self._call_llm(prompt)
