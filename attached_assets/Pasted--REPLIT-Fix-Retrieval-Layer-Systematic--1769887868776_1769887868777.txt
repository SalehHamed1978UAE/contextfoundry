  def get_relationship_types_for_role(role_query: str) -> list:                                          
      """Map a role query to KG relationship types."""                                                   
      role_lower = role_query.lower()                                                                    
      for role_term, rel_types in ROLE_TO_RELATIONSHIP_MAP.items():                                      
          if role_term in role_lower:                                                                    
              return rel_types                                                                           
      return ['HOLDS_POSITION', 'MANAGES', 'LEADS']  # Default fallback

  def get_customers_for_product(product_name: str) -> list:                                              
      """                                                                                                
      Get customers for a product.                                                                       
      If no direct CUSTOMER_OF edges, traverse through parent org.                                       
      """                                                                                                
      # First try direct: Product ← CUSTOMER_OF ← Customer                                               
      direct_customers = query_kg(f"""                                                                   
          SELECT c.name FROM entities c                                                                  
          JOIN relationships r ON c.id = r.source_id                                                     
          JOIN entities p ON r.target_id = p.id                                                          
          WHERE p.name ILIKE '%{product_name}%'                                                          
          AND r.relationship_type = 'CUSTOMER_OF'                                                        
      """)                                                                                               
                                                                                                         
      if direct_customers:                                                                               
          return direct_customers                                                                        
                                                                                                         
      # Fallback: Product → PRODUCT_OF → Division ← CUSTOMER_OF ← Customer                               
      indirect_customers = query_kg(f"""                                                                 
          SELECT c.name FROM entities c                                                                  
          JOIN relationships r1 ON c.id = r1.source_id  -- Customer → CUSTOMER_OF                        
          JOIN entities div ON r1.target_id = div.id    -- → Division                                    
          JOIN relationships r2 ON div.id = r2.target_id -- Division ← PRODUCT_OF                        
          JOIN entities p ON r2.source_id = p.id        -- ← Product                                     
          WHERE p.name ILIKE '%{product_name}%'                                                          
          AND r1.relationship_type = 'CUSTOMER_OF'                                                       
          AND r2.relationship_type IN ('PRODUCT_OF', 'PART_OF', 'BELONGS_TO')                            
      """)                                                                                               
                                                                                                         
      return indirect_customers
