```python
async def resolve_disambiguation(
    query: str, 
    matches: List[Dict], 
    vault_context: str
) -> Dict:
    """Resolve ambiguous matches using LLM reasoning."""
    
    # Fast path: single match = no reasoning needed
    if len(matches) == 1:
        return {
            "single_answer": True,
            "primary_result": matches[0],
            "alternatives": []
        }
    
    # Fast path: exact vault name match = obvious answer
    exact_matches = [m for m in matches if vault_context.lower() in (m.get('org') or '').lower()]
    if len(exact_matches) == 1:
        return {
            "single_answer": True,
            "primary_result": exact_matches[0],
            "alternatives": [m for m in matches if m != exact_matches[0]]
        }
    
    # Complex case: let LLM reason
    return await llm_reason_about_matches(query, matches, vault_context)


async def llm_reason_about_matches(
    query: str,
    matches: List[Dict],
    vault_context: str
) -> Dict:
    """Use LLM to reason about which match(es) best answer the user's intent."""
    
    # Format matches with their relationships
    matches_description = "\n".join([
        f"- {m['name']} ({m['role']}) → works at: {m.get('organizations', ['Unknown'])}"
        for m in matches
    ])
    
    prompt = f"""The user asked: "{query}"
Vault context: "{vault_context}"

I found these matches in the knowledge graph:
{matches_description}

Reason about which match(es) best answer the user's intent:
1. Consider the vault context - what organization is this vault about?
2. Consider the relationships - who works where?
3. Determine if there's a clear primary answer or if multiple are relevant

Respond in JSON only:
{{
    "reasoning": "brief explanation of your analysis",
    "single_answer": true if one clear best answer else false,
    "primary_result": {{"name": "...", "role": "...", "org": "..."}},
    "show_alternatives": true if user should see other options,
    "alternatives": [{{"name": "...", "role": "...", "org": "..."}}]
}}"""
    
    response = await llm.complete(prompt, response_format="json")
    return json.loads(response)
```

================================================================================
KEY PRINCIPLES
================================================================================

1. Graph provides FACTS (who holds what role, who works where)
2. LLM provides REASONING (which fact answers the user's intent)
3. Fast paths for obvious cases (single match, exact vault match)
4. LLM only called when genuinely ambiguous
5. Response is structured JSON, not free-form text

This is the general solution for ALL disambiguation - roles, entities, projects, departments.
