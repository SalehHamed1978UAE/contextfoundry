Then add enforcement:
sqlCREATE OR REPLACE FUNCTION validate_entity_type()
RETURNS TRIGGER AS $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM ontology.types 
        WHERE LOWER(type_name) = LOWER(NEW.entity_type) 
        AND status = 'ACTIVE'
    ) THEN
        RAISE EXCEPTION 'Invalid entity_type: %. Must match ACTIVE type in ontology.types', NEW.entity_type;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER enforce_entity_type
BEFORE INSERT OR UPDATE ON entities
FOR EACH ROW EXECUTE FUNCTION validate_entity_type();
