PART 3: API ENDPOINT
═══════════════════════════════════════════════════════════════

Add to web_app.py:

@app.route('/api/documents/<doc_id>/details')
def document_details(doc_id):
    """Get document details including all extracted entities."""
    tenant_id = session.get('tenant_id')
    
    # Get document
    doc = db.execute("""
        SELECT id, name, status, file_type, file_size, created_at,
               extraction_method, extraction_metrics, published
        FROM platform.documents 
        WHERE id = %s AND tenant_id = %s
    """, [doc_id, tenant_id]).fetchone()
    
    if not doc:
        return jsonify({'error': 'Document not found'}), 404
    
    # Get entities grouped by type
    entities = db.execute("""
        SELECT entity_type, name, confidence
        FROM public.entities
        WHERE source_document_id = %s AND tenant_id = %s
        ORDER BY entity_type, confidence DESC, name
    """, [doc_id, tenant_id]).fetchall()
    
    # Group entities by type
    grouped = {}
    for e in entities:
        etype = e['entity_type']
        if etype not in grouped:
            grouped[etype] = []
        grouped[etype].append({
            'name': e['name'],
            'confidence': float(e['confidence']) if e['confidence'] else 0.9
        })
    
    return jsonify({
        'id': str(doc['id']),
        'name': doc['name'],
        'status': doc['status'],
        'file_type': doc['file_type'],
        'file_size': doc['file_size'],
        'created_at': doc['created_at'].isoformat() if doc['created_at'] else None,
        'extraction_method': doc['extraction_method'] or 'text',
        'extraction_metrics': doc['extraction_metrics'] or {},
        'published': doc['published'] or False,
        'entity_count': len(entities),
        'entities_by_type': grouped
    })

PART 4: JAVASCRIPT

let currentDocumentId = null;

function openDetailPanel(docId) {
    currentDocumentId = docId;
    
    // Fetch document details
    fetch(`/api/documents/${docId}/details`)
        .then(r => r.json())
        .then(data => {
            // Populate header
            document.getElementById('panel-doc-name').textContent = data.name;
            
            // Populate metadata
            document.getElementById('panel-doc-type').textContent = data.file_type?.toUpperCase() || 'Unknown';
            document.getElementById('panel-doc-size').textContent = formatFileSize(data.file_size);
            document.getElementById('panel-doc-date').textContent = formatDate(data.created_at);
            document.getElementById('panel-doc-method').textContent = data.extraction_method || 'text';
            document.getElementById('panel-doc-status').textContent = data.published ? 'Published' : 'Review';
            document.getElementById('panel-doc-status').className = 'meta-value badge ' + (data.published ? 'published' : 'review');
            
            // Domain from metrics
            const metrics = data.extraction_metrics || {};
            document.getElementById('panel-doc-domain').textContent = 
                metrics.domain ? `${metrics.domain} (${metrics.domain_confidence?.toFixed(2) || '?'})` : 'Core';
            
            // Populate entities
            document.getElementById('panel-entity-count').textContent = data.entity_count;
            
            const entityList = document.getElementById('panel-entity-list');
            entityList.innerHTML = '';
            
            for (const [type, entities] of Object.entries(data.entities_by_type)) {
                const group = document.createElement('div');
                group.className = 'entity-group';
                group.innerHTML = `
                    <div class="entity-type-header">
                        ${type} <span class="entity-type-count">(${entities.length})</span>
                    </div>
                `;
                
                for (const entity of entities) {
                    const item = document.createElement('div');
                    item.className = 'entity-item';
                    item.innerHTML = `
                        <span>${entity.name}</span>
                        <span class="entity-confidence">${(entity.confidence * 100).toFixed(0)}%</span>
                    `;
                    group.appendChild(item);
                }
                
                entityList.appendChild(group);
            }
            
            // Show panel
            document.getElementById('document-detail-panel').classList.add('open');
            document.getElementById('panel-overlay').classList.add('open');
        })
        .catch(err => {
            console.error('Failed to load document details:', err);
            alert('Failed to load document details');
        });
}

function closeDetailPanel() {
    document.getElementById('document-detail-panel').classList.remove('open');
    document.getElementById('panel-overlay').classList.remove('open');
    currentDocumentId = null;
}

function formatFileSize(bytes) {
    if (!bytes) return 'Unknown';
    if (bytes < 1024) return bytes + ' B';
    if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
    return (bytes / (1024 * 1024)).toFixed(1) + ' MB';
}

// Make document rows clickable
document.addEventListener('click', function(e) {
    const row = e.target.closest('tr[data-doc-id]');
    if (row) {
        openDetailPanel(row.dataset.docId);
    }
});
