@app.route('/api/documents/<doc_id>/publish', methods=['POST'])
def publish_document(doc_id):
    """Approve document - entities become trusted."""
    tenant_id = session.get('tenant_id')
    if not tenant_id:
        return jsonify({'error': 'Not authenticated'}), 401
    
    conn = get_db_connection()
    cur = conn.cursor()
    try:
        # Update document
        cur.execute("""
            UPDATE platform.documents 
            SET published = TRUE, published_at = NOW(), published_by = %s
            WHERE id = %s AND tenant_id = %s
            RETURNING id
        """, [session.get('user_email', 'user'), doc_id, tenant_id])
        
        if cur.fetchone() is None:
            return jsonify({'error': 'Document not found'}), 404
        
        # Update entities to trusted
        cur.execute("""
            UPDATE public.entities 
            SET status = 'trusted'
            WHERE source_document_id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        
        conn.commit()
        return jsonify({'success': True, 'published': True})
    finally:
        cur.close()
        conn.close()

@app.route('/api/documents/<doc_id>/reject', methods=['POST'])
def reject_document(doc_id):
    """Reject document - delete entities, mark as rejected."""
    tenant_id = session.get('tenant_id')
    if not tenant_id:
        return jsonify({'error': 'Not authenticated'}), 401
    
    conn = get_db_connection()
    cur = conn.cursor()
    try:
        # Delete relationships first (foreign key constraint)
        cur.execute("""
            DELETE FROM public.relationships 
            WHERE tenant_id = %s AND (
                source_entity_id IN (SELECT id FROM public.entities WHERE source_document_id = %s)
                OR target_entity_id IN (SELECT id FROM public.entities WHERE source_document_id = %s)
            )
        """, [tenant_id, doc_id, doc_id])
        
        # Delete entities
        cur.execute("""
            DELETE FROM public.entities 
            WHERE source_document_id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        
        # Mark document as rejected
        cur.execute("""
            UPDATE platform.documents 
            SET status = 'rejected', published = FALSE
            WHERE id = %s AND tenant_id = %s
            RETURNING id
        """, [doc_id, tenant_id])
        
        if cur.fetchone() is None:
            return jsonify({'error': 'Document not found'}), 404
        
        conn.commit()
        return jsonify({'success': True, 'status': 'rejected'})
    finally:
        cur.close()
        conn.close()


@app.route('/api/documents/<doc_id>/re-extract', methods=['POST'])
def re_extract_document(doc_id):
    """Re-extract document - delete entities, re-queue for extraction."""
    tenant_id = session.get('tenant_id')
    if not tenant_id:
        return jsonify({'error': 'Not authenticated'}), 401
    
    conn = get_db_connection()
    cur = conn.cursor()
    try:
        # Get document info
        cur.execute("""
            SELECT id, name, file_type, file_size 
            FROM platform.documents 
            WHERE id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        doc = cur.fetchone()
        
        if not doc:
            return jsonify({'error': 'Document not found'}), 404
        
        # Delete existing relationships
        cur.execute("""
            DELETE FROM public.relationships 
            WHERE tenant_id = %s AND (
                source_entity_id IN (SELECT id FROM public.entities WHERE source_document_id = %s)
                OR target_entity_id IN (SELECT id FROM public.entities WHERE source_document_id = %s)
            )
        """, [tenant_id, doc_id, doc_id])
        
        # Delete existing entities
        cur.execute("""
            DELETE FROM public.entities 
            WHERE source_document_id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        
        # Reset document status
        cur.execute("""
            UPDATE platform.documents 
            SET status = 'queued', published = FALSE, 
                extraction_method = NULL, extraction_metrics = NULL
            WHERE id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        
        # Create new extraction request
        cur.execute("""
            INSERT INTO platform.extraction_requests 
            (id, request_id, document_id, tenant_id, file_path, file_name, 
             mime_type, file_size_bytes, extraction_mode, priority, status, 
             retry_count, max_retries, created_at, submitted_at)
            VALUES (
                gen_random_uuid(), gen_random_uuid(), %s, %s,
                './storage/tenants/' || %s || '/documents/' || %s || '/v1/content',
                %s, %s, %s, 'full', 'high', 'pending', 0, 3, NOW(), NOW()
            )
        """, [doc_id, tenant_id, tenant_id, doc_id, 
              doc[1], 'application/pdf', doc[3]])
        
        conn.commit()
        return jsonify({'success': True, 'status': 'queued'})
    finally:
        cur.close()
        conn.close()


@app.route('/api/documents/<doc_id>', methods=['DELETE'])
def delete_document(doc_id):
    """Hard delete document and all associated data."""
    tenant_id = session.get('tenant_id')
    if not tenant_id:
        return jsonify({'error': 'Not authenticated'}), 401
    
    conn = get_db_connection()
    cur = conn.cursor()
    try:
        # Delete relationships
        cur.execute("""
            DELETE FROM public.relationships 
            WHERE tenant_id = %s AND (
                source_entity_id IN (SELECT id FROM public.entities WHERE source_document_id = %s)
                OR target_entity_id IN (SELECT id FROM public.entities WHERE source_document_id = %s)
            )
        """, [tenant_id, doc_id, doc_id])
        
        # Delete entities
        cur.execute("""
            DELETE FROM public.entities 
            WHERE source_document_id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        
        # Delete extraction requests
        cur.execute("""
            DELETE FROM platform.extraction_requests 
            WHERE document_id = %s AND tenant_id = %s
        """, [doc_id, tenant_id])
        
        # Delete document
        cur.execute("""
            DELETE FROM platform.documents 
            WHERE id = %s AND tenant_id = %s
            RETURNING id
        """, [doc_id, tenant_id])
        
        if cur.fetchone() is None:
            return jsonify({'error': 'Document not found'}), 404
        
        conn.commit()
        return jsonify({'success': True, 'deleted': True})
    finally:
        cur.close()
        conn.close()

function publishDocument() {
    if (!currentDocumentId) return;
    
    fetch(`/api/documents/${currentDocumentId}/publish`, { method: 'POST' })

function getStatusBadge(doc) {
    if (doc.status === 'rejected') {
        return '<span class="badge badge-rejected">⚫ Rejected</span>';
    } else if (doc.status === 'extracted') {
        if (doc.published) {
            return '<span class="badge badge-published">✅ Published</span>';
        } else {
            return '<span class="badge badge-review">🟡 Review</span>';
        }
    } else if (doc.status === 'processing') {
        return '<span class="badge badge-processing">⏳ Processing</span>';
    } else if (doc.status === 'queued') {
        return '<span class="badge badge-queued">📋 Queued</span>';
    } else if (doc.status === 'failed') {
        return '<span class="badge badge-failed">❌ Failed</span>';
    }
    return '<span class="badge">' + doc.status + '</span>';
}
