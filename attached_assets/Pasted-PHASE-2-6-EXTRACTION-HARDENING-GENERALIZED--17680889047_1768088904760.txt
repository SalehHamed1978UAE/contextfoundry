def _extract_person_from_role_pattern(self, text: str, role_query: str) -> Optional[str]:
    """Find person for a SPECIFIC role, not any C-suite role."""
    
    # Expand abbreviations to all variants
    role_expansions = {
        'ceo': ['CEO', 'Chief Executive Officer'],
        'cfo': ['CFO', 'Chief Financial Officer'],
        'cto': ['CTO', 'Chief Technology Officer'],
        'cio': ['CIO', 'Chief Information Officer'],
        'coo': ['COO', 'Chief Operating Officer'],
        'cdo': ['CDO', 'Chief Data Officer'],
        'cmo': ['CMO', 'Chief Medical Officer', 'Chief Marketing Officer'],
        'cno': ['CNO', 'Chief Nursing Officer'],
        'managing partner': ['Managing Partner'],
        'senior partner': ['Senior Partner'],
    }
    
    role_lower = role_query.lower().strip()
    role_variants = role_expansions.get(role_lower, [role_query])
    
    for role_variant in role_variants:
        role_escaped = re.escape(role_variant)
        
        # Patterns for THIS SPECIFIC ROLE only
        patterns = [
            # "Thomas Bradley - Chief Financial Officer"
            rf"([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s*[-–—:]\s*{role_escaped}",
            # "Chief Financial Officer Thomas Bradley"
            rf"{role_escaped}\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)",
            # "Thomas Bradley, CFO"
            rf"([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+),\s*{role_escaped}",
            # With prefix: "Dr. Margaret Chen - CEO"
            rf"(?:Dr\.|Mr\.|Ms\.|Mrs\.)\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s*[-–—:]\s*{role_escaped}",
        ]
        
        for pattern in patterns:
            match = re.search(pattern, text, re.IGNORECASE)
            if match:
                full_name = match.group(1).strip()
                logger.info(f"[ROLE_RESOLVER] Stage 3: '{role_variant}' → '{full_name}'")
                return full_name
    
    return None

```python
import re
import yaml
from typing import List, Tuple, Set, Dict
from pathlib import Path

class ExtractionPostProcessor:
    """
    Pattern-based post-processing for ALL known relationship types.
    Config-driven — add patterns without code changes.
    """
    
    def __init__(self, session, tenant_id: str, config_path: str = None):
        self.session = session
        self.tenant_id = tenant_id
        self.patterns = self._load_patterns(config_path)
    
    def _load_patterns(self, config_path: str) -> Dict:
        """Load relationship patterns from config."""
        if config_path and Path(config_path).exists():
            with open(config_path) as f:
                return yaml.safe_load(f)
        
        # Default patterns if no config
        return {
            'HOLDS_POSITION': {
                'patterns': [
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s*[-–—:]\s*(Chief\s+\w+\s+Officer|CEO|CFO|CTO|CIO|COO|CDO|CMO|CNO|VP\s+\w+|Director\s+\w+|Managing\s+Partner|Senior\s+Partner|Partner|Chair\w*)',
                    r'(CEO|CFO|CTO|CIO|COO)\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)',
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+),\s*(CEO|CFO|CTO|CIO|COO|Managing\s+Partner)',
                ],
                'source_type': 'PERSON',
                'target_type': 'ROLE',
            },
            'REPORTS_TO': {
                'patterns': [
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s+reports\s+to\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)',
                    r'Reports\s+to:\s*([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)',
                ],
                'source_type': 'PERSON',
                'target_type': 'PERSON',
            },
            'INVESTED_IN': {
                'patterns': [
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)\s+invested\s+.*?in\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)',
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)\s+has\s+(?:a\s+)?\d+%\s+(?:stake|ownership)\s+in\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)',
                ],
                'source_type': 'ORGANIZATION',
                'target_type': 'ORGANIZATION',
            },
            'WORKS_AT': {
                'patterns': [
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s+(?:works|employed)\s+at\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)',
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s+joined\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)',
                ],
                'source_type': 'PERSON',
                'target_type': 'ORGANIZATION',
            },
            'LEADS': {
                'patterns': [
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s+leads?\s+(?:the\s+)?([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)',
                    r'([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\s*[-–]\s*(?:Head|Chair|Lead)\s+(?:of\s+)?([A-Z][a-z]+)',
                ],
                'source_type': 'PERSON',
                'target_type': 'DEPARTMENT',
            },
        }
    
    def process(self, text: str, entities: List, relationships: List) -> Tuple[List, List]:
        """
        Find relationships the LLM missed across ALL known types.
        """
        for rel_type, config in self.patterns.items():
            found = self._find_patterns(text, config['patterns'])
            
            for source_name, target_name in found:
                if not self._relationship_exists(source_name, target_name, rel_type, relationships):
                    # Create entities if needed
                    source_entity = self._get_or_create_entity(
                        source_name, config['source_type'], entities
                    )
                    target_entity = self._get_or_create_entity(
                        target_name, config['target_type'], entities
                    )
                    
                    # Create relationship
                    new_rel = self._create_relationship(source_entity, rel_type, target_entity)
                    relationships.append(new_rel)
                    
                    logger.info(f"[POST-PROCESSOR] Added: {source_name} -[{rel_type}]-> {target_name}")
        
        return entities, relationships
    
    def _find_patterns(self, text: str, patterns: List[str]) -> Set[Tuple[str, str]]:
        """Find all matches for given patterns."""
        found = set()
        for pattern in patterns:
            for match in re.finditer(pattern, text, re.IGNORECASE):
                groups = match.groups()
                if len(groups) >= 2:
                    found.add((groups[0].strip(), groups[1].strip()))
        return found
    
    def _relationship_exists(self, source: str, target: str, rel_type: str, relationships: List) -> bool:
        """Check if relationship already exists (fuzzy match)."""
        source_lower = source.lower()
        target_lower = target.lower()
        for rel in relationships:
            if rel.get('relationship_type') != rel_type:
                continue
            rel_source = rel.get('source_name', '').lower()
            rel_target = rel.get('target_name', '').lower()
            if (source_lower in rel_source or rel_source in source_lower) and \
               (target_lower in rel_target or rel_target in target_lower):
                return True
        return False
    
    def _get_or_create_entity(self, name: str, entity_type: str, entities: List):
        """Get existing entity or create new one."""
        for entity in entities:
            if entity.get('name', '').lower() == name.lower():
                return entity
        new_entity = {'name': name, 'entity_type': entity_type}
        entities.append(new_entity)
        return new_entity
    
    def _create_relationship(self, source, rel_type: str, target) -> dict:
        """Create relationship dict."""
        return {
            'source_name': source.get('name') if isinstance(source, dict) else source,
            'relationship_type': rel_type,
            'target_name': target.get('name') if isinstance(target, dict) else target,
        }
```

class GapSeverity(Enum):
    HIGH = "HIGH"
    MEDIUM = "MEDIUM"
    LOW = "LOW"

@dataclass

    gap_type: str
    severity: GapSeverity
    message: str
    details: Dict

class ExtractionValidator:
    """
    Gap detection for ALL known relationship types.
    Config-driven — extend without code changes.
    """
    
    # Keywords that suggest a relationship should exist
    RELATIONSHIP_INDICATORS = {
        'HOLDS_POSITION': {

        'WORKS_AT': {
            'keywords': ['works at', 'employed', 'employee', 'joined', 
                        'staff', 'team member'],
            'entity_type': 'PERSON',
        },
    }
    
    def __init__(self, session, tenant_id: str):
        self.session = session
        self.tenant_id = tenant_id
    
    def validate(self, document_id: str) -> List[ExtractionGap]:
        """Validate extraction for all known relationship types."""
        gaps = []
        doc_text = self._get_document_text(document_id).lower()
        
        for rel_type, config in self.RELATIONSHIP_INDICATORS.items():
            # Check if document has keywords suggesting this relationship

            if has_keywords:
                rel_count = self._count_relationships(document_id, rel_type)

                    gaps.append(ExtractionGap(
                        gap_type=f'MISSING_{rel_type}',

                        message=f"Document has {rel_type} keywords but no {rel_type} relationships",

                gap_type='PEOPLE_WITHOUT_ROLES',
                severity=GapSeverity.HIGH,
                message=f"Found {person_count} people but no role relationships",
                details={'person_count': person_count, 'role_count': role_count}

def extract_and_process(self, document_id: str, text: str):
    # Step 1: LLM extraction (existing)
    entities, relationships = self._llm_extract(text)
    logger.info(f"[EXTRACTION] LLM extracted: {len(entities)} entities, {len(relationships)} relationships")
    
    # Step 2: Post-processing (NEW)
    post_processor = ExtractionPostProcessor(self.session, self.tenant_id)
    entities, relationships = post_processor.process(text, entities, relationships)
    logger.info(f"[EXTRACTION] After post-processing: {len(entities)} entities, {len(relationships)} relationships")
    
    # Step 3: Save to database (existing)
    self._save_to_database(entities, relationships)
    
    # Step 4: Validation (NEW)
    validator = ExtractionValidator(self.session, self.tenant_id)
    gaps = validator.validate(document_id)
    
    if gaps:
        logger.warning(f"[EXTRACTION] {len(gaps)} gaps detected")
        for gap in gaps:
            logger.warning(f"  - {gap.severity.value}: {gap.message}")
    
    return entities, relationships, gaps
