def classify_domain(content: str) -> str:
    """Classify document into a domain based on content."""
    prompt = f"""Classify this document into ONE domain.

Document:
{content[:2000]}

Domains:
- it_infrastructure: servers, databases, deployments, incidents, APIs
- healthcare: patients, diagnoses, procedures, medications, hospitals
- aviation: flights, aircraft, airports, crew, routes
- finance: bonds, equities, transactions, accounts, portfolios
- supply_chain: suppliers, shipments, inventory, warehouses
- manufacturing: products, machines, factories, quality control
- construction: projects, sites, contractors, materials, permits
- general: none of the above

Reply with ONLY the domain code, nothing else."""

    response = call_llm(prompt)
    return response.strip().lower()
