pythondef parse_question_bank(md_file: Path) -> List[dict]:
    """Parse markdown question bank into list of {id, question, expected}"""
    content = md_file.read_text()
    questions = []
    
    # Pattern: **Q1:** question text\n**A1:** answer text
    pattern = r'\*\*Q(\d+):\*\*\s*(.*?)\n\*\*A\1:\*\*\s*(.*?)(?=\n\*\*Q|\Z)'
    
    for match in re.finditer(pattern, content, re.DOTALL):
        questions.append({
            "id": int(match.group(1)),
            "question": match.group(2).strip(),
            "expected": match.group(3).strip()
        })
    
    return questions
