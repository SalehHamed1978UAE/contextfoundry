        # Override the test field
        if field_name == 'normalized_name' and value == "":

            store.add_relationship_candidate(**kwargs)
            
            if should_succeed:
                print(f"    ✓ {field_name}={repr(value)}: Accepted (expected)")

        ("EMPLOYED_BY", "WORKS_AT"),
        ("INVESTS_IN", "INVESTED_IN"),

            print(f"    ✗ {input_type} → {normalized} (expected {expected_normalized})")

    # 18c: Test routing through loader
    print("\n  18c. Routing through StagingLoader:")

        ("WORKS_AT", "kg"),  # Known
        ("EMPLOYED_BY", "kg"),  # Normalizes to known
        ("COMPLETELY_UNKNOWN_TYPE_XYZ", "candidate"),  # Unknown
    ]
    
    for rel_type, expected_action in test_cases:
        test_marker = f"ROUTE_{uuid4().hex[:4].upper()}"

            result = loader.load_relation(
                tenant_id=TENANT_ID,
                relation_data={
                    'source_entity': 'TestSource',

                    'target_entity': 'TestTarget',
                    'target_type': 'ORGANIZATION',
                    'relationship_type': rel_type,
                    'properties': {},
                    'document_id': uuid4(),
                    'chunk_id': uuid4(),

    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    all_passed = True
    
    test_type = f"PROV_{uuid4().hex[:4].upper()}"
    doc_id = uuid4()
    chunk_id = uuid4()
    
    store.add_relationship_candidate(
        proposed_name=test_type,
        normalized_name=test_type,
        source_entity_type="ORGANIZATION",
        target_entity_type="PERSON",
        source_entity_name="Acme Corp",
        target_entity_name="John Smith",
        properties={"role": "CEO", "since": "2020"},
        original_text="John Smith has been CEO of Acme Corp since 2020.",
        document_id=doc_id,
        chunk_id=chunk_id
    )
    
    candidates = store.get_pending_candidates()
    candidate = next((c for c in candidates if c['normalized_name'] == test_type), None)
    
    if not candidate:
        print(f"    ✗ Candidate not found")
        return False
    
    candidate_id = UUID(candidate['id'])
    extractions = store.get_pending_extractions(candidate_id)
    
    if not extractions:
        print(f"    ✗ No extractions found")
        return False
    
    ext = extractions[0]
    
    # Check all provenance fields
    checks = [
        ('source_entity_name', 'Acme Corp'),
        ('target_entity_name', 'John Smith'),
        ('source_entity_type', 'ORGANIZATION'),
        ('target_entity_type', 'PERSON'),
        ('relationship_type', test_type),
    ]
    
    for field, expected in checks:
        actual = ext.get(field)
        if actual == expected:
            print(f"    ✓ {field}: {actual}")
        else:
            print(f"    ✗ {field}: {actual} (expected {expected})")
            all_passed = False
    
    # Check document_id
    if ext.get('document_id'):
        print(f"    ✓ document_id: present")
    else:
        print(f"    ✗ document_id: missing")
        all_passed = False
    
    # Check properties
    props = ext.get('properties', {})
    if isinstance(props, str):
        props = json.loads(props)
    
    if props.get('role') == 'CEO':
        print(f"    ✓ properties preserved: {props}")
    else:
        print(f"    ✗ properties not preserved: {props}")
        all_passed = False
    
    # Check chunk_text
    chunk_text = ext.get('chunk_text', '')
    if 'CEO of Acme Corp' in chunk_text:
        print(f"    ✓ chunk_text preserved")
    else:
        print(f"    ✗ chunk_text not preserved: {chunk_text[:50]}...")
        all_passed = False
    
    # Cleanup
    session.execute(text(f"DELETE FROM pending_extractions WHERE candidate_id IN (SELECT id FROM ontology_candidates WHERE normalized_name = '{test_type}')"))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{test_type}'"))

                'PERF_%', 'INTEGRITY_%', 'RECOVERY_%', 'LEN_%', 'PROP_%',
                'NULL_%', 'TYPE_%', 'CASE_%', 'ROUTE_%', 'PROV_%', 
                'STATS_%', 'FILTER_%', 'TEST_%', 'CONF_%', 'CAP_%']
    
    total_found = 0
    
    for pattern in patterns:
        result = session.execute(text(f"""
            SELECT COUNT(*) as count FROM ontology_candidates 
            WHERE normalized_name LIKE '{pattern}'
        """)).fetchone()
        
        if result.count > 0:
            print(f"    ⚠️  {pattern}: {result.count} candidates remaining")
            total_found += result.count
    
    if total_found == 0:
        print(f"    ✓ All test data cleaned up")
        return True
    else:
        print(f"\n    ✗ {total_found} test candidates remain - cleaning up...")
        
        for pattern in patterns:
            session.execute(text(f"""
                DELETE FROM pending_extractions 
                WHERE candidate_id IN (
                    SELECT id FROM ontology_candidates WHERE normalized_name LIKE '{pattern}'
                )
            """))
            session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name LIKE '{pattern}'"))
        
        session.commit()
        print(f"    ✓ Cleaned up")
        return True
