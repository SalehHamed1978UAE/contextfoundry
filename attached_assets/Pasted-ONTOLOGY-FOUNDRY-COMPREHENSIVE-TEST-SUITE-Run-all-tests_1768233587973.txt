def test_2_confidence_increase():
    """Test that confidence increases with more mentions across documents."""
    print("\n" + "=" * 70)
    print("TEST 2: CONFIDENCE INCREASE WITH MULTIPLE MENTIONS")
    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    
    test_type = f"CONFIDENCE_TEST_{uuid4().hex[:8].upper()}"
    
    confidence_history = []
    
    # Add mentions from different documents
    for i in range(5):
        doc_id = uuid4()
        store.add_relationship_candidate(
            proposed_name=test_type,
            normalized_name=test_type,
            source_entity_type="ORGANIZATION",
            target_entity_type="ORGANIZATION",
            source_entity_name=f"Company{chr(65+i)}",
            target_entity_name=f"Company{chr(66+i)}",
            properties={},
            original_text=f"Company{chr(65+i)} {test_type.lower()} Company{chr(66+i)}",
            document_id=doc_id
        )
        
        # Get current confidence
        candidates = store.get_pending_candidates()
        candidate = next((c for c in candidates if c['normalized_name'] == test_type), None)
        
        if candidate:
            confidence_history.append({
                'docs': candidate['document_count'],
                'mentions': candidate['mention_count'],
                'confidence': candidate['confidence_score']
            })
            print(f"  After {i+1} docs: confidence = {candidate['confidence_score']:.2f} "
                  f"({candidate['document_count']} docs, {candidate['mention_count']} mentions)")
    
    # Verify confidence increased
    if len(confidence_history) >= 2:
        first = confidence_history[0]['confidence']
        last = confidence_history[-1]['confidence']
        increased = last > first
        print(f"\n  Confidence increased: {first:.2f} → {last:.2f} {'✓' if increased else '✗'}")
    
    # Cleanup
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (
            SELECT id FROM ontology_candidates WHERE normalized_name = '{test_type}'
        )
    """))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{test_type}'"))
    session.commit()
    print("  Cleaned up test data")
    
    return len(confidence_history) >= 3 and confidence_history[-1]['confidence'] > confidence_history[0]['confidence']

def test_3_normalizer_coverage():
    """Test that synonyms normalize correctly."""
    print("\n" + "=" * 70)
    print("TEST 3: NORMALIZER SYNONYM COVERAGE")
    print("=" * 70)
    
    normalizer = CandidateNormalizer()
    
    test_cases = [
        # Investment patterns
        ("INVESTS_IN", "INVESTED_IN"),
        ("MADE_INVESTMENT_IN", "INVESTED_IN"),
        ("FUNDED", "INVESTED_IN"),
        ("BACKS", "INVESTED_IN"),
        ("LED_ROUND_IN", "INVESTED_IN"),
        
        # Acquisition patterns
        ("BOUGHT", "ACQUIRED"),
        ("PURCHASED", "ACQUIRED"),
        ("TOOK_OVER", "ACQUIRED"),
        
        # Employment patterns
        ("EMPLOYED_BY", "WORKS_AT"),
        ("EMPLOYEE_OF", "WORKS_AT"),
        ("WORKS_FOR", "WORKS_AT"),
        ("JOINED", "WORKS_AT"),
        
        # Ownership patterns
        ("OWNED_BY", "OWNS"),
        ("SUBSIDIARY_OF", "OWNS"),
        ("PARENT_OF", "OWNS"),
        ("CONTROLS", "OWNS"),
        
        # Leadership patterns
        ("LEADS", "HOLDS_POSITION"),
        ("MANAGES", "HOLDS_POSITION"),
        ("HEADS", "HOLDS_POSITION"),
        ("IS_CEO_OF", "HOLDS_POSITION"),
        
        # Board patterns
        ("ON_BOARD_OF", "BOARD_MEMBER_OF"),
        ("BOARD_DIRECTOR_OF", "BOARD_MEMBER_OF"),
        ("SITS_ON_BOARD", "BOARD_MEMBER_OF"),
        
        # Partnership patterns
        ("PARTNERS_WITH", "PARTNERED_WITH"),
        ("COLLABORATES_WITH", "PARTNERED_WITH"),
        ("ALLIED_WITH", "PARTNERED_WITH"),
        
        # Unknown (should pass through unchanged)
        ("COMPLETELY_NOVEL_TYPE", "COMPLETELY_NOVEL_TYPE"),
        ("MENTORS", "MENTORS"),
        ("ADVISES", "ADVISES"),
    ]
    
    passed = 0
    failed = 0
    
    for input_name, expected in test_cases:
        result = normalizer.normalize_relationship(input_name)
        if result == expected:
            passed += 1
            print(f"  ✓ {input_name} → {result}")
        else:
            failed += 1
            print(f"  ✗ {input_name} → {result} (expected {expected})")
    
    print(f"\n  Passed: {passed}/{len(test_cases)}")
    
    # Also test case insensitivity and cleaning
    print("\n  Testing case insensitivity and cleaning:")
    edge_cases = [
        ("invests_in", "INVESTED_IN"),
        ("Invests-In", "INVESTED_IN"),
        ("  INVESTS  IN  ", "INVESTS_IN"),  # Should clean but may not map
    ]
    
    for input_name, expected in edge_cases:
        result = normalizer.normalize_relationship(input_name)
        status = "✓" if result == expected else "~"
        print(f"  {status} '{input_name}' → '{result}'")
    
    return failed == 0

def test_5_pipeline_trace():
    """Trace the full flow with detailed logging."""
    print("\n" + "=" * 70)
    print("TEST 5: FULL PIPELINE TRACE")
    print("=" * 70)
    
    session = get_session()
    loader = StagingLoader(session)
    store = CandidateStore(session, TENANT_ID)
    
    test_type = f"PIPELINE_TRACE_{uuid4().hex[:8].upper()}"
    document_id = uuid4()
    chunk_id = uuid4()
    
    print(f"\n  Test relationship type: {test_type}")
    print(f"  Document ID: {document_id}")
    
    # Step 1: Check known types
    known_types = loader._get_known_relationship_types()
    is_known = test_type in known_types
    print(f"\n  Step 1: Check known types")
    print(f"    Known types count: {len(known_types)}")
    print(f"    Is '{test_type}' known? {is_known}")
    
    # Step 2: Normalize
    normalizer = CandidateNormalizer()
    normalized = normalizer.normalize_relationship(test_type)
    print(f"\n  Step 2: Normalize")
    print(f"    '{test_type}' → '{normalized}'")
    
    # Step 3: Route decision
    print(f"\n  Step 3: Route decision")
    if is_known:
        print(f"    → Would write to KG")
    else:
        print(f"    → Would store as candidate")
    
    # Step 4: Execute
    print(f"\n  Step 4: Execute")
    relation_data = {
        'source_entity': 'TraceCompanyA',
        'source_type': 'ORGANIZATION',
        'target_entity': 'TraceCompanyB',
        'target_type': 'ORGANIZATION',
        'relationship_type': test_type,
        'properties': {'trace': 'test'},
        'document_id': document_id,
        'chunk_id': chunk_id,
        'chunk_text': f'TraceCompanyA {test_type.lower()} TraceCompanyB for testing.'
    }
    
    try:
        result = loader.load_relation(tenant_id=TENANT_ID, relation_data=relation_data)
        print(f"    Result: {result}")
    except Exception as e:
        print(f"    Error: {e}")
        return False
    
    # Step 5: Verify storage
    print(f"\n  Step 5: Verify storage")
    candidates = store.get_pending_candidates()
    candidate = next((c for c in candidates if c['normalized_name'] == normalized), None)
    
    if candidate:
        print(f"    ✓ Candidate found:")
        print(f"      ID: {candidate['id']}")
        print(f"      Confidence: {candidate['confidence_score']:.2f}")
        print(f"      Documents: {candidate['document_count']}")
        
        # Check pending extractions
        extractions = store.get_pending_extractions(UUID(candidate['id']))
        print(f"      Pending extractions: {len(extractions)}")
        if extractions:
            print(f"      First extraction: {extractions[0]['source_entity_name']} → {extractions[0]['target_entity_name']}")
    else:
        print(f"    ✗ Candidate not found")
        return False
    
    # Cleanup
    print(f"\n  Step 6: Cleanup")
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (SELECT id FROM ontology_candidates WHERE normalized_name = '{normalized}')
    """))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{normalized}'"))
    session.commit()
    print(f"    ✓ Test data cleaned up")
    
    return True
