def test_24_camelcase_regression():
    """Verify CamelCase fix works for all patterns."""
    print("\n" + "=" * 70)
    print("TEST 24: CAMELCASE NORMALIZATION (REGRESSION)")
    print("=" * 70)
    
    normalizer = CandidateNormalizer()
    all_passed = True
    
    test_cases = [
        # Basic CamelCase
        ("WorksAt", "WORKS_AT"),
        ("ReportsTo", "REPORTS_TO"),
        ("InvestedIn", "INVESTED_IN"),
        ("PartneredWith", "PARTNERED_WITH"),
        
        # Multiple humps
        ("WorksAtCompany", "WORKS_AT_COMPANY"),
        ("IsEmployedByOrganization", "IS_EMPLOYED_BY_ORGANIZATION"),
        
        # Acronyms in CamelCase
        ("WorksAtIBM", "WORKS_AT_IBM"),
        ("CEOOf", "CEO_OF"),
        ("CTOAt", "CTO_AT"),
        
        # Mixed patterns
        ("worksAt", "WORKS_AT"),  # lowerCamelCase
        ("WORKS_AT", "WORKS_AT"),  # Already uppercase
        ("works_at", "WORKS_AT"),  # Already snake_case
        
        # Numbers
        ("Founded2020", "FOUNDED_2020"),
        ("Series_A_Investor", "SERIES_A_INVESTOR"),
        
        # Edge cases
        ("A", "A"),
        ("AB", "AB"),
        ("ABC", "ABC"),
        ("ABCDef", "ABC_DEF"),
        ("AbCdEf", "AB_CD_EF"),
    ]
    
    for input_val, expected in test_cases:
        result = normalizer.normalize_relationship(input_val)
        if result == expected:
            print(f"  ✓ '{input_val}' → '{result}'")
        else:
            print(f"  ✗ '{input_val}' → '{result}' (expected '{expected}')")
            all_passed = False
    
    return all_passed

        document_id=uuid4()

        proposed_name=test_type_b,
        normalized_name=test_type_b,
        source_entity_type="ORGANIZATION",
        target_entity_type="ORGANIZATION",
        source_entity_name="X",
        target_entity_name="Y",
        original_text="Tenant B data",
        document_id=uuid4()
    )
    
    # Tenant A should only see their candidate
    candidates_a = store_a.get_pending_candidates()
    sees_own = any(c['normalized_name'] == test_type_a for c in candidates_a)
    sees_other = any(c['normalized_name'] == test_type_b for c in candidates_a)
    
    if sees_own and not sees_other:
        print(f"  ✓ Tenant A sees only their data")
    else:
        print(f"  ✗ Tenant A isolation failed (sees_own={sees_own}, sees_other={sees_other})")
        all_passed = False
    
    # Tenant B should only see their candidate
    candidates_b = store_b.get_pending_candidates()
    sees_own = any(c['normalized_name'] == test_type_b for c in candidates_b)
    sees_other = any(c['normalized_name'] == test_type_a for c in candidates_b)
    
    if sees_own and not sees_other:
        print(f"  ✓ Tenant B sees only their data")

        print(f"  ✗ Tenant B isolation failed")

    # Summary
    print("\n" + "=" * 70)
    print("DEEPER EDGE CASE TEST SUMMARY")
    print("=" * 70)
    
    passed = sum(1 for v in results.values() if v)
    total = len(results)
    
    for test_name, passed_test in results.items():
        status = "✓ PASSED" if passed_test else "✗ FAILED"
        print(f"  {status}: {test_name}")
    
    print(f"\n  Total: {passed}/{total} tests passed")
    
    if passed == total:
        print("\n  ✅ ALL DEEPER TESTS PASSED")
    else:
        print("\n  ⚠️  Some tests failed")
    
    print("=" * 70)

if __name__ == "__main__":
    main()
