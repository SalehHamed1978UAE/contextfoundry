@app.route('/api/corpus-stats')
def corpus_stats():
    tenant_id = session.get('tenant_id') or g.tenant_id
    
    doc_count = db.execute(
        "SELECT COUNT(*) FROM platform.documents WHERE tenant_id = %s",
        [tenant_id]
    ).fetchone()[0]
    
    processing_count = db.execute(
        "SELECT COUNT(*) FROM platform.documents WHERE tenant_id = %s AND status IN ('queued', 'processing')",
        [tenant_id]
    ).fetchone()[0]
    
    entity_count = db.execute(
        "SELECT COUNT(*) FROM public.entities WHERE tenant_id = %s",
        [tenant_id]
    ).fetchone()[0]
    
    relationship_count = db.execute(
        "SELECT COUNT(*) FROM public.relationships WHERE tenant_id = %s",
        [tenant_id]
    ).fetchone()[0]
    
    return jsonify({
        'documents': doc_count,
        'entities': entity_count,
        'relationships': relationship_count,
        'processing': processing_count
    })

import pytesseract
from pdf2image import convert_from_path

def extract_text_from_pdf(file_path: str) -> str:
    """Extract text from PDF, with OCR fallback for scanned docs."""
    
    # Try regular text extraction first
    try:
        from pypdf import PdfReader
        reader = PdfReader(file_path)
        text = ""
        for page in reader.pages:
            page_text = page.extract_text() or ""
            text += page_text
        
        # If we got meaningful text, return it
        if len(text.strip()) > 100:
            return text
    except Exception as e:
        print(f"pypdf extraction failed: {e}")
    
    # Try pdfplumber
    try:
        import pdfplumber
        with pdfplumber.open(file_path) as pdf:
            text = ""
            for page in pdf.pages:
                page_text = page.extract_text() or ""
                text += page_text
            
            if len(text.strip()) > 100:
                return text
    except Exception as e:
        print(f"pdfplumber extraction failed: {e}")
    
    # OCR fallback for scanned PDFs
    try:
        print(f"Using OCR for scanned PDF: {file_path}")
        images = convert_from_path(file_path, dpi=200, first_page=1, last_page=20)  # Limit pages for performance
        text = ""
        for i, image in enumerate(images):
            page_text = pytesseract.image_to_string(image)
            text += f"\n--- Page {i+1} ---\n{page_text}"
        return text
    except Exception as e:
        print(f"OCR extraction failed: {e}")
        return ""
