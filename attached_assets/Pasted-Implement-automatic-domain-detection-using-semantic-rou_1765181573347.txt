STEP 2: Create classify_document function

def classify_document(text_sample: str) -> tuple[str, float]:
    """Classify document using semantic routing."""
    
    # Embed first 1000 chars of document
    sample = text_sample[:1000]
    doc_embedding = get_embedding(sample)
    
    # Cosine similarity against all domain embeddings
    scores = {}
    for domain, domain_embedding in DOMAIN_EMBEDDINGS.items():
        scores[domain] = cosine_similarity(doc_embedding, domain_embedding)
    
    best_domain = max(scores, key=scores.get)
    best_score = scores[best_domain]
    
    # Return domain only if high confidence
    if best_score >= 0.85:
        return best_domain, best_score
    else:
        return 'core', best_score

def extract_entities(document_text: str, tenant_id: str):
    # 1. Classify document (cheap embedding call)
    domain, confidence = classify_document(document_text)
    
    # 2. Build composite type list (Core + Domain)
    entity_types = CORE_TYPES.copy()
    if domain != 'core':
        entity_types += DOMAIN_TYPES.get(domain, [])
    
    # Cap at 25 types max
    entity_types = entity_types[:25]
    
    # 3. Single extraction call
    entities, relationships = run_extraction(document_text, entity_types)
    
    # 4. Log for telemetry
    print(f"Document classified as: {domain} (confidence: {confidence:.2f})")
    print(f"Types used: {entity_types}")
    print(f"Extracted: {len(entities)} entities")
    
    return entities, relationships
