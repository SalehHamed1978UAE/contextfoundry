def test_35_round_trip():
    """Complete round-trip test with all fields."""
    print("\n" + "=" * 70)
    print("TEST 35: FULL ROUND-TRIP")
    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    all_passed = True
    
    # Create with all fields populated
    test_type = f"ROUND_{uuid4().hex[:4].upper()}"
    doc_id = uuid4()
    chunk_id = uuid4()
    
    input_data = {
        'proposed_name': f"Original_{test_type}",
        'normalized_name': test_type,
        'source_entity_type': "PERSON",
        'target_entity_type': "ORGANIZATION",
        'source_entity_name': "John Smith",
        'target_entity_name': "Acme Corporation",
        'properties': {"role": "CEO", "since": 2020, "active": True},
        'original_text': "John Smith became CEO of Acme Corporation in 2020.",
        'document_id': doc_id,
        'chunk_id': chunk_id
    }
    
    candidate_id = store.add_relationship_candidate(**input_data)
    print(f"  Created candidate: {candidate_id}")
    
    # Retrieve candidate
    candidate = store.get_candidate_by_id(candidate_id)
    
    if not candidate:
        print(f"  ✗ Failed to retrieve candidate")
        return False
    
    # Verify candidate fields
    checks = [
        ('normalized_name', test_type),
        ('candidate_type', 'RELATIONSHIP'),
        ('status', 'PENDING'),
        ('source_entity_type', 'PERSON'),
        ('target_entity_type', 'ORGANIZATION'),
        ('document_count', 1),
        ('mention_count', 1),
    ]
    
    for field, expected in checks:
        actual = candidate.get(field)
        if actual == expected:
            print(f"  ✓ {field}: {actual}")
        else:
            print(f"  ✗ {field}: {actual} (expected {expected})")
            all_passed = False
    
    # Verify example mentions
    mentions = candidate.get('example_mentions', [])
    if mentions and 'CEO of Acme' in mentions[0].get('text', ''):
        print(f"  ✓ example_mentions: text preserved")
    else:
        print(f"  ✗ example_mentions: text not preserved")
        all_passed = False
    
    # Retrieve and verify extraction
    extractions = store.get_pending_extractions(candidate_id)
    
    if not extractions:
        print(f"  ✗ No extractions found")
        return False
    
    ext = extractions[0]
    ext_checks = [
        ('source_entity_name', 'John Smith'),
        ('target_entity_name', 'Acme Corporation'),
        ('source_entity_type', 'PERSON'),
        ('target_entity_type', 'ORGANIZATION'),
        ('relationship_type', test_type),
        ('status', 'PENDING'),
    ]
    
    for field, expected in ext_checks:
        actual = ext.get(field)
        if actual == expected:
            print(f"  ✓ extraction.{field}: {actual}")
        else:
            print(f"  ✗ extraction.{field}: {actual} (expected {expected})")
            all_passed = False
    
    # Verify properties preserved
    props = ext.get('properties', {})
    if isinstance(props, str):
        props = json.loads(props)
    
    if props.get('role') == 'CEO' and props.get('since') == 2020:
        print(f"  ✓ extraction.properties: preserved")
    else:
        print(f"  ✗ extraction.properties: not preserved ({props})")
        all_passed = False
    
    # Cleanup
    session.execute(text(f"DELETE FROM pending_extractions WHERE candidate_id = '{candidate_id}'"))

def test_40_confidence_formula():
    """Verify confidence formula: 40% doc diversity + 30% mention freq + 30% base."""
    print("\n" + "=" * 70)
    print("TEST 40: CONFIDENCE FORMULA VERIFICATION")
    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    all_passed = True
    
    # Test cases: (docs, mentions, expected_confidence)
    # Formula: min(docs/3, 1)*0.4 + min(mentions/5, 1)*0.3 + 0.3
    test_cases = [
        (1, 1, 0.49),   # 0.13 + 0.06 + 0.30 = 0.49
        (2, 2, 0.69),   # 0.27 + 0.12 + 0.30 = 0.69
        (3, 3, 0.88),   # 0.40 + 0.18 + 0.30 = 0.88
        (3, 5, 1.00),   # 0.40 + 0.30 + 0.30 = 1.00
        (5, 10, 1.00),  # Capped at 1.0
    ]
    
    for docs, mentions, expected in test_cases:
        calculated = store._calculate_confidence(docs, mentions)
        
        # Allow small floating point differences
        if abs(calculated - expected) < 0.02:
            print(f"  ✓ ({docs} docs, {mentions} mentions) → {calculated:.2f} (expected {expected:.2f})")
        else:
            print(f"  ✗ ({docs} docs, {mentions} mentions) → {calculated:.2f} (expected {expected:.2f})")
            all_passed = False
    
    return all_passed
