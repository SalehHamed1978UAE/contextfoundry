FIX 3: Chunker Table Preservation (1 hour)

The chunker splits markdown tables mid-row. Tables contain structured facts (leadership teams, financial data) that get lost.

```python
# File to edit: src/context_foundry/chunking/chunker.py (or wherever chunking lives)
# If chunker.py doesn't exist, find the chunking code in parsing/document_parser.py
# or wherever text is split into chunks

import re

def chunk_with_table_preservation(text, chunk_size=1500, overlap=100):
    """
    Split text into chunks while preserving markdown tables as atomic units.
    A table row is never split across two chunks.
    """
    # Pattern: markdown table rows (lines starting with |)
    table_row_pattern = r'^\|.*\|$'
    
    lines = text.split('\n')
    chunks = []
    current_chunk = []
    current_size = 0
    pending_table = []  # Buffer for complete table
    in_table = False
    
    for line in lines:
        is_table_row = re.match(table_row_pattern, line.strip())
        
        if is_table_row:
            in_table = True
            pending_table.append(line)
            continue
        elif in_table:
            # We were in a table, now we're not. Flush the table.
            in_table = False
            table_text = '\n'.join(pending_table)
            pending_table = []
            
            # If table fits in current chunk, add it
            if current_size + len(table_text) <= chunk_size:
                current_chunk.append(table_text)
                current_size += len(table_text)
            else:
                # Table doesn't fit — start new chunk with just the table
                if current_chunk:
                    chunks.append('\n'.join(current_chunk))
                current_chunk = [table_text]
                current_size = len(table_text)
        
        # Add non-table line
        if current_size + len(line) > chunk_size and current_chunk:
            chunks.append('\n'.join(current_chunk))
            # Keep overlap
            overlap_text = '\n'.join(current_chunk)[-overlap:]
            current_chunk = [overlap_text, line] if overlap_text else [line]
            current_size = len('\n'.join(current_chunk))
        else:
            current_chunk.append(line)
            current_size += len(line) + 1
    
    # Flush remaining
    if current_chunk:
        chunks.append('\n'.join(current_chunk))
    if pending_table:
        table_text = '\n'.join(pending_table)
        if current_chunk:
            chunks[-1] += '\n' + table_text
        else:
            chunks.append(table_text)
    
    return chunks
```
