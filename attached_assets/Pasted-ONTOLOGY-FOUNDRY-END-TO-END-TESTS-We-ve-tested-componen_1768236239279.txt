    known_relationships = [
        ("John Smith", "PERSON", "Acme Corp", "ORGANIZATION", "WORKS_AT"),

        ("Bob Wilson", "PERSON", "CEO", "JOB_TITLE", "HOLDS_POSITION"),
    ]
    
    print(f"\n  Processing {len(known_relationships)} known-type relationships...")
    
    # We need to go through the actual pipeline
    # Import the staging loader or extraction pipeline
    try:
        from src.context_foundry.staging.staging_loader import StagingLoader
        from src.context_foundry.extraction.models import ExtractedRelation
        
        loader = StagingLoader(session)
        
        for source, source_type, target, target_type, rel_type in known_relationships:
            # Create ExtractedRelation object (adapt to your actual model)
            relation = ExtractedRelation(
                source_entity=source,
                source_type=source_type,
                target_entity=target,
                target_type=target_type,
                relationship_type=rel_type,
                properties={},
                confidence=0.9
            )
            
            # Process through loader
            result = loader.load_relation(relation, tenant_id=TENANT_ID)
            action = result.get('action', 'unknown') if isinstance(result, dict) else str(result)
            print(f"    {rel_type}: {action}")

        print(f"  ⚠️  Could not import pipeline components: {e}")
        print(f"  Testing via direct candidate store instead...")

    doc_id = uuid4()

        store.add_relationship_candidate(
            proposed_name=rel_type,
            normalized_name=rel_type,
            source_entity_type=source_type,
            target_entity_type=target_type,
            source_entity_name=source,
            target_entity_name=target,
            properties={},
            original_text=f"{source} {rel_type.lower()} {target}",
            document_id=doc_id
        )
        print(f"    {rel_type}: → candidate")
    
    # Verify candidates created
    after_count = len(store.get_pending_candidates())
    new_candidates = after_count - before_count
    
    print(f"\n  Candidates after: {after_count}")
    print(f"  New candidates: {new_candidates}")
    
    # Verify each type exists
    candidates = store.get_pending_candidates()
    found_types = [c['normalized_name'] for c in candidates]
    
    all_found = True
    for _, _, _, _, rel_type in unknown_relationships:
        if rel_type in found_types:
            print(f"    ✓ {rel_type} found")
        else:
            print(f"    ✗ {rel_type} NOT found")

        session.execute(text(f"""

        print(f"\n  ✓ All unknown types became candidates")

def test_e2e_mixed_types():
    """
    E2E: Document with both known and unknown types.
    Expected: Known → KG, Unknown → candidates.
    """
    print("\n" + "=" * 70)
    print("E2E TEST 3: DOCUMENT WITH MIXED TYPES")
    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    normalizer = CandidateNormalizer()
    
    # Get known types from the system
    try:
        from src.context_foundry.staging.staging_loader import StagingLoader
        loader = StagingLoader(session)
        known_types = loader._get_known_relationship_types()
    except:
        known_types = {'WORKS_AT', 'INVESTED_IN', 'HOLDS_POSITION', 'FOUNDED', 'LOCATED_IN'}
    
    print(f"  Known types in system: {len(known_types)}")
    
    # Get baseline
    before_count = len(store.get_pending_candidates())
    
    test_id = uuid4().hex[:4].upper()
    doc_id = uuid4()
    
    # Mixed relationships
    relationships = [
        # Known types (should go to KG)
        ("Alice", "PERSON", "TechCorp", "ORGANIZATION", "WORKS_AT", True),
        
        # Unknown types (should become candidates)
        ("BigCo", "ORGANIZATION", "SmallCo", "ORGANIZATION", f"E2E_ACQUIRED_{test_id}", False),
        ("VC Fund", "ORGANIZATION", "Startup", "ORGANIZATION", f"E2E_SPONSORS_{test_id}", False),
    ]
    
    kg_count = 0
    candidate_count = 0
    
    for source, source_type, target, target_type, rel_type, is_known in relationships:
        normalized = normalizer.normalize_relationship(rel_type)
        should_be_known = normalized in known_types
        
        if should_be_known:
            print(f"    {rel_type} → {normalized} → KG (known)")
            kg_count += 1
        else:
            # Add as candidate
            store.add_relationship_candidate(
                proposed_name=rel_type,
                normalized_name=normalized,
                source_entity_type=source_type,
                target_entity_type=target_type,
                source_entity_name=source,
                target_entity_name=target,
                original_text=f"{source} {rel_type} {target}",
                document_id=doc_id
            )
            print(f"    {rel_type} → {normalized} → Candidate")
            candidate_count += 1
    
    # Verify
    after_count = len(store.get_pending_candidates())
    new_candidates = after_count - before_count
    
    print(f"\n  Summary:")
    print(f"    Routed to KG: {kg_count}")
    print(f"    Routed to candidates: {candidate_count}")
    print(f"    New candidates in DB: {new_candidates}")
    
    # Cleanup
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (
            SELECT id FROM ontology_candidates WHERE normalized_name LIKE 'E2E_%{test_id}'
        )
    """))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name LIKE 'E2E_%{test_id}'"))
    session.commit()
    
    if new_candidates == candidate_count:
        print(f"\n  ✓ Mixed routing working correctly")
        return True
    else:
        print(f"\n  ✗ Routing mismatch")
        return False

def test_e2e_multiple_docs():
    """
    E2E: Multiple documents mentioning the same unknown type.
    Expected: Single candidate with increasing confidence.
    """
    print("\n" + "=" * 70)
    print("E2E TEST 4: MULTIPLE DOCUMENTS SAME UNKNOWN TYPE")
    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    
    test_type = f"E2E_MULTI_{uuid4().hex[:4].upper()}"
    num_docs = 5
    
    print(f"\n  Creating {num_docs} documents with type: {test_type}")
    
    confidence_history = []
    
    for i in range(num_docs):
        doc_id = uuid4()
        
        store.add_relationship_candidate(
            proposed_name=test_type,
            normalized_name=test_type,
            source_entity_type="ORGANIZATION",
            target_entity_type="ORGANIZATION",
            source_entity_name=f"Company{i}",
            target_entity_name=f"Target{i}",
            original_text=f"Document {i}: Company{i} {test_type} Target{i}",
            document_id=doc_id
        )
        
        # Check confidence after each doc
        candidates = store.get_pending_candidates()
        candidate = next((c for c in candidates if c['normalized_name'] == test_type), None)
        
        if candidate:
            confidence_history.append({
                'doc': i + 1,
                'confidence': candidate['confidence_score'],
                'docs': candidate['document_count'],
                'mentions': candidate['mention_count']
            })
            print(f"    Doc {i+1}: confidence={candidate['confidence_score']:.2f}, "
                  f"docs={candidate['document_count']}, mentions={candidate['mention_count']}")
    
    # Verify confidence increased
    if len(confidence_history) >= 2:
        first = confidence_history[0]['confidence']
        last = confidence_history[-1]['confidence']
        increased = last > first
        
        print(f"\n  Confidence progression: {first:.2f} → {last:.2f}")
        
        if increased:
            print(f"  ✓ Confidence increased with more documents")
        else:
            print(f"  ✗ Confidence did not increase")
    
    # Verify single candidate
    candidates = store.get_pending_candidates()
    matching = [c for c in candidates if c['normalized_name'] == test_type]
    
    if len(matching) == 1:
        print(f"  ✓ Single candidate (not duplicated)")
    else:
        print(f"  ✗ Expected 1 candidate, found {len(matching)}")
    
    # Verify extraction count
    if matching:
        candidate_id = UUID(matching[0]['id'])
        extractions = store.get_pending_extractions(candidate_id)
        
        if len(extractions) == num_docs:
            print(f"  ✓ All {num_docs} extractions stored")
        else:
            print(f"  ✗ Expected {num_docs} extractions, found {len(extractions)}")
    
    # Cleanup
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (
            SELECT id FROM ontology_candidates WHERE normalized_name = '{test_type}'
        )
    """))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{test_type}'"))
    session.commit()
    
    return (len(matching) == 1 and 
            len(confidence_history) == num_docs and 
            confidence_history[-1]['confidence'] > confidence_history[0]['confidence'])

def test_e2e_synonyms():
    """
    E2E: Different synonyms should normalize to same candidate.
    """
    print("\n" + "=" * 70)
    print("E2E TEST 5: SYNONYM NORMALIZATION")
    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    normalizer = CandidateNormalizer()
    
    # Test synonym group
    synonyms = ["INVESTS_IN", "FUNDED", "BACKS", "MADE_INVESTMENT_IN"]
    canonical = "INVESTED_IN"
    
    print(f"\n  Testing synonyms → {canonical}")
    
    # Check if canonical is known
    try:
        from src.context_foundry.staging.staging_loader import StagingLoader
        loader = StagingLoader(session)
        known_types = loader._get_known_relationship_types()
        is_known = canonical in known_types
    except:
        is_known = True  # Assume known for this test
    
    print(f"  {canonical} is known: {is_known}")
    
    if is_known:
        print(f"\n  Since {canonical} is known, synonyms should route to KG:")
        for syn in synonyms:
            normalized = normalizer.normalize_relationship(syn)
            if normalized == canonical:
                print(f"    ✓ {syn} → {normalized} → KG")
            else:
                print(f"    ✗ {syn} → {normalized} (expected {canonical})")
        
        return True
    else:
        # If not known, all synonyms should create ONE candidate
        print(f"\n  Since {canonical} is unknown, synonyms should create ONE candidate:")
        
        before_count = len(store.get_pending_candidates())
        
        for i, syn in enumerate(synonyms):
            normalized = normalizer.normalize_relationship(syn)
            store.add_relationship_candidate(
                proposed_name=syn,
                normalized_name=normalized,
                source_entity_type="ORGANIZATION",
                target_entity_type="ORGANIZATION",
                source_entity_name=f"Investor{i}",
                target_entity_name=f"Company{i}",
                original_text=f"Investor{i} {syn} Company{i}",
                document_id=uuid4()
            )
            print(f"    {syn} → {normalized}")
        
        # Should be single candidate
        candidates = store.get_pending_candidates()
        matching = [c for c in candidates if c['normalized_name'] == canonical]
        
        if len(matching) == 1:
            print(f"\n  ✓ All synonyms merged into single candidate")
            print(f"    Mentions: {matching[0]['mention_count']}")
        else:
            print(f"\n  ✗ Expected 1 candidate, found {len(matching)}")
        
        # Cleanup
        session.execute(text(f"""
            DELETE FROM pending_extractions 
            WHERE candidate_id IN (
                SELECT id FROM ontology_candidates WHERE normalized_name = '{canonical}'
            )
        """))
        session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{canonical}'"))
        session.commit()
        
        return len(matching) == 1

def test_e2e_full_trace():

    Trace complete flow with detailed logging.
    """

    print("=" * 70)
    
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    normalizer = CandidateNormalizer()
    
    test_type = f"E2E_TRACE_{uuid4().hex[:4].upper()}"
    doc_id = uuid4()
    chunk_id = uuid4()
    
    print(f"\n  Step 1: Input")
    print(f"    Relationship type: {test_type}")
    print(f"    Document ID: {doc_id}")
    print(f"    Chunk ID: {chunk_id}")
    
    print(f"\n  Step 2: Normalization")
    normalized = normalizer.normalize_relationship(test_type)
    print(f"    {test_type} → {normalized}")
    
    print(f"\n  Step 3: Known type check")
    try:
        from src.context_foundry.staging.staging_loader import StagingLoader
        loader = StagingLoader(session)
        known_types = loader._get_known_relationship_types()
        is_known = normalized in known_types
    except:
        is_known = False
    print(f"    Is known: {is_known}")
    
    print(f"\n  Step 4: Routing decision")
    if is_known:
        print(f"    → Route to Knowledge Graph")
    else:
        print(f"    → Route to Candidate Store")
    
    print(f"\n  Step 5: Storage")
    if not is_known:
        candidate_id = store.add_relationship_candidate(
            proposed_name=test_type,
            normalized_name=normalized,
            source_entity_type="ORGANIZATION",
            target_entity_type="ORGANIZATION",
            source_entity_name="TraceOrg",
            target_entity_name="TraceTarget",
            properties={"trace": True, "test_id": test_type},
            original_text=f"TraceOrg {test_type} TraceTarget for E2E testing.",
            document_id=doc_id,
            chunk_id=chunk_id
        )
        print(f"    Candidate ID: {candidate_id}")
    
    print(f"\n  Step 6: Verification")
    candidates = store.get_pending_candidates()
    candidate = next((c for c in candidates if c['normalized_name'] == normalized), None)
    
    if candidate:
        print(f"    ✓ Candidate found")
        print(f"      ID: {candidate['id']}")
        print(f"      Confidence: {candidate['confidence_score']:.2f}")
        print(f"      Status: {candidate['status']}")
        
        # Check extraction
        candidate_id = UUID(candidate['id'])
        extractions = store.get_pending_extractions(candidate_id)
        
        if extractions:
            ext = extractions[0]
            print(f"    ✓ Extraction found")
            print(f"      Source: {ext['source_entity_name']}")
            print(f"      Target: {ext['target_entity_name']}")
            print(f"      Status: {ext['status']}")
        else:
            print(f"    ✗ No extraction found")
    else:
        if is_known:
            print(f"    ✓ No candidate (type is known, went to KG)")
        else:
            print(f"    ✗ Candidate not found")
    
    # Cleanup
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (
            SELECT id FROM ontology_candidates WHERE normalized_name = '{normalized}'
        )
    """))
    session.execute(text(f"DELETE FROM ontology_candidates WHERE normalized_name = '{normalized}'"))
    session.commit()
    print(f"\n  Step 7: Cleanup complete")
    
    return True

            source_entity_name=f"Org{i}",
            target_entity_name=f"Target{i}",
            original_text=f"Test {i}",

        )
    
    # Check stats
    after_stats = store.get_stats()
    after_pending = after_stats.get('RELATIONSHIP', {}).get('PENDING', 0)
    
    diff = after_pending - before_pending
    
    print(f"  After: {after_pending} pending relationships")
    print(f"  Difference: +{diff}")
    
    if diff == num_to_create:
        print(f"  ✓ Stats accurate")
        success = True
    else:
        print(f"  ✗ Expected +{num_to_create}, got +{diff}")
        success = False
    
    # Cleanup
    session.execute(text(f"""
        DELETE FROM pending_extractions 
        WHERE candidate_id IN (
            SELECT id FROM ontology_candidates WHERE normalized_name LIKE '{test_prefix}%'
        )
    """))

    final_stats = store.get_stats()
    final_pending = final_stats.get('RELATIONSHIP', {}).get('PENDING', 0)

    setup()
    
    results = {}
    
    tests = [
        ("e2e_known_types", test_e2e_known_types),
        ("e2e_unknown_types", test_e2e_unknown_types),
        ("e2e_mixed_types", test_e2e_mixed_types),
        ("e2e_multiple_docs", test_e2e_multiple_docs),
        ("e2e_synonyms", test_e2e_synonyms),
        ("e2e_existing_test", test_e2e_existing_test),
        ("e2e_full_trace", test_e2e_full_trace),
        ("e2e_stats", test_e2e_stats),
    ]
    
    for name, func in tests:
        try:
            results[name] = func()
        except Exception as e:
            print(f"  ✗ CRASHED: {e}")
            import traceback
            traceback.print_exc()
            results[name] = False
    
    # Summary
    print("\n" + "=" * 70)
    print("E2E TEST SUMMARY")
    print("=" * 70)
    
    passed = sum(1 for v in results.values() if v)
    total = len(results)
    
    for test_name, passed_test in results.items():
        status = "✓ PASSED" if passed_test else "✗ FAILED"
        print(f"  {status}: {test_name}")
    
    print(f"\n  Total: {passed}/{total} E2E tests passed")
    
    # Grand total
    print("\n" + "=" * 70)
    print("COMPLETE TEST SUMMARY")
    print("=" * 70)
    print("  Basic Tests:      6/6")
    print("  Stress Tests:     5/6")
    print("  Exhaustive Tests: 6/6")
    print("  Deeper Tests:     8/8")
    print("  Final Tests:      5/6")
    print(f"  E2E Tests:        {passed}/{total}")
    print(f"  ─────────────────────")
    print(f"  GRAND TOTAL:      {30 + passed}/{32 + total}")
    
    if passed == total:
        print("\n  ✅ ALL E2E TESTS PASSED — READY FOR PHASE 2")
    else:
        print("\n  ⚠️  Some E2E tests failed — Review before Phase 2")
    
    print("=" * 70)

if __name__ == "__main__":
    main()
