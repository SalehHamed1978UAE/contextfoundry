Instructions for Replit: Extraction Monitoring Dashboard                                               
                                                                                                         
  Overview                                                                                               
                                                                                                         
  Add an "Extraction" section to the admin dashboard that shows:                                         
  - Queue status across all vaults                                                                       
  - Per-vault progress with document-level detail                                                        
  - Event log with restart/failure history                                                               
  - Action buttons for control                                                                           
                                                                                                         
  ---                                                                                                    
  Part 1: Database Schema                                                                                
                                                                                                         
  1.1 Add Extraction Timestamps to Documents                                                             
                                                                                                         
  -- Add columns to track extraction completion times                                                    
  ALTER TABLE platform.documents                                                                         
  ADD COLUMN IF NOT EXISTS extraction_level VARCHAR(20) DEFAULT NULL,                                    
  ADD COLUMN IF NOT EXISTS single_extracted_at TIMESTAMP,                                                
  ADD COLUMN IF NOT EXISTS multi_extracted_at TIMESTAMP;                                                 
                                                                                                         
  1.2 Create Extraction Events Log Table                                                                 
                                                                                                         
  CREATE TABLE IF NOT EXISTS platform.extraction_events (                                                
      id SERIAL PRIMARY KEY,                                                                             
      vault_id UUID NOT NULL,                                                                            
      vault_name VARCHAR(255),                                                                           
      document_id UUID,                                                                                  
      document_name VARCHAR(255),                                                                        
      event_type VARCHAR(50) NOT NULL,  -- 'started', 'resumed', 'completed', 'failed', 'cancelled'      
      extraction_level VARCHAR(20),      -- 'single', 'multi'                                            
      details TEXT,                       -- Additional context or error message                         
      created_at TIMESTAMP DEFAULT NOW()                                                                 
  );                                                                                                     
                                                                                                         
  CREATE INDEX idx_extraction_events_vault ON platform.extraction_events(vault_id);                      
  CREATE INDEX idx_extraction_events_created ON platform.extraction_events(created_at DESC);             
  CREATE INDEX idx_extraction_events_type ON platform.extraction_events(event_type);

  @app.route('/api/extraction/overview')                                                                 
  def get_extraction_overview():                                                                         
      """Get extraction status across all vaults with aggregate metrics."""                              
      with get_db_session() as session:                                                                  
          vaults = TenantService().list_tenants()                                                        
                                                                                                         
          vault_stats = []                                                                               
          totals = {                                                                                     
              "total_vaults": 0,                                                                         
              "vaults_pending": 0,                                                                       
              "vaults_in_progress": 0,                                                                   
              "vaults_complete": 0,                                                                      
              "total_docs": 0,                                                                           
              "docs_pending_multi": 0                                                                    
          }                                                                                              
                                                                                                         
          for vault in vaults:                                                                           
              stats = session.execute(text("""                                                           
                  SELECT                                                                                 
                      COUNT(*) as total,                                                                 
                      COUNT(*) FILTER (WHERE extraction_level = 'multi') as multi_done,                  
                      COUNT(*) FILTER (WHERE extraction_level = 'single') as single_done,                
                      COUNT(*) FILTER (WHERE extraction_level IS NULL OR extraction_level != 'multi') as 
  pending_multi,                                                                                         
                      COUNT(*) FILTER (WHERE status = 'failed') as failed,                               
                      COUNT(*) FILTER (WHERE status = 'processing') as in_progress,                      
                      MAX(multi_extracted_at) as last_multi_at,                                          
                      MAX(single_extracted_at) as last_single_at                                         
                  FROM platform.documents                                                                
                  WHERE tenant_id = :vault_id                                                            
              """), {"vault_id": vault.id}).fetchone()                                                   
                                                                                                         
              # Determine vault status                                                                   
              if stats.in_progress > 0:                                                                  
                  vault_status = "in_progress"                                                           
                  totals["vaults_in_progress"] += 1                                                      
              elif stats.multi_done == stats.total and stats.total > 0:                                  
                  vault_status = "complete"                                                              
                  totals["vaults_complete"] += 1                                                         
              elif stats.total == 0:                                                                     
                  vault_status = "empty"                                                                 
              else:                                                                                      
                  vault_status = "pending"                                                               
                  totals["vaults_pending"] += 1                                                          
                                                                                                         
              totals["total_vaults"] += 1                                                                
              totals["total_docs"] += stats.total                                                        
              totals["docs_pending_multi"] += stats.pending_multi                                        
                                                                                                         
              vault_stats.append({                                                                       
                  "vault_id": str(vault.id),                                                             
                  "vault_name": vault.name,                                                              
                  "status": vault_status,                                                                
                  "total_docs": stats.total,                                                             
                  "single_done": stats.single_done,                                                      
                  "multi_done": stats.multi_done,                                                        
                  "pending_multi": stats.pending_multi,                                                  
                  "failed": stats.failed,                                                                
                  "in_progress": stats.in_progress,                                                      
                  "progress_pct": round((stats.multi_done / stats.total * 100), 1) if stats.total > 0    
  else 0,                                                                                                
                  "last_updated": (stats.last_multi_at or stats.last_single_at or "").isoformat() if     
  (stats.last_multi_at or stats.last_single_at) else None                                                
              })                                                                                         
                                                                                                         
          return jsonify({                                                                               
              "totals": totals,                                                                          
              "vaults": sorted(vault_stats, key=lambda x: x["pending_multi"], reverse=True)              
          })

  @app.route('/api/extraction/vault/<vault_id>')                                                         
  def get_vault_extraction_detail(vault_id):                                                             
      """Get detailed extraction status for each document in a vault."""                                 
      with get_db_session() as session:                                                                  
          docs = session.execute(text("""                                                                
              SELECT                                                                                     
                  id,                                                                                    
                  name,                                                                                  
                  status,                                                                                
                  extraction_level,                                                                      
                  single_extracted_at,                                                                   
                  multi_extracted_at,                                                                    
                  created_at                                                                             
              FROM platform.documents                                                                    
              WHERE tenant_id = :vault_id                                                                
              ORDER BY                                                                                   
                  CASE                                                                                   
                      WHEN status = 'failed' THEN 0                                                      
                      WHEN status = 'processing' THEN 1                                                  
                      WHEN extraction_level IS NULL THEN 2                                               
                      WHEN extraction_level = 'single' THEN 3                                            
                      ELSE 4                                                                             
                  END,                                                                                   
                  name                                                                                   
          """), {"vault_id": vault_id}).fetchall()                                                       
                                                                                                         
          vault = TenantService().get_tenant(vault_id)                                                   
                                                                                                         
          return jsonify({                                                                               
              "vault_id": vault_id,                                                                      
              "vault_name": vault.name if vault else "Unknown",                                          
              "documents": [{                                                                            
                  "id": str(d.id),                                                                       
                  "name": d.name,                                                                        
                  "status": d.status,                                                                    
                  "extraction_level": d.extraction_level or "pending",                                   
                  "single_completed_at": d.single_extracted_at.isoformat() if d.single_extracted_at else 
  None,                                                                                                  
                  "multi_completed_at": d.multi_extracted_at.isoformat() if d.multi_extracted_at else    
  None,                                                                                                  
                  "uploaded_at": d.created_at.isoformat() if d.created_at else None                      
              } for d in docs]                                                                           
          })

  @app.route('/api/extraction/events')                                                                   
  @app.route('/api/extraction/events/<vault_id>')                                                        
  def get_extraction_events(vault_id=None):                                                              
      """Get extraction event history with filtering."""                                                 
      limit = request.args.get('limit', 100, type=int)                                                   
      event_type = request.args.get('type')  # Optional filter: 'failed', 'resumed', etc.                
                                                                                                         
      with get_db_session() as session:                                                                  
          query = """                                                                                    
              SELECT vault_id, vault_name, document_id, document_name,                                   
                     event_type, extraction_level, details, created_at                                   
              FROM platform.extraction_events                                                            
              WHERE 1=1                                                                                  
          """                                                                                            
          params = {"limit": limit}                                                                      
                                                                                                         
          if vault_id:                                                                                   
              query += " AND vault_id = :vault_id"                                                       
              params["vault_id"] = vault_id                                                              
                                                                                                         
          if event_type:                                                                                 
              query += " AND event_type = :event_type"                                                   
              params["event_type"] = event_type                                                          
                                                                                                         
          query += " ORDER BY created_at DESC LIMIT :limit"                                              
                                                                                                         
          rows = session.execute(text(query), params).fetchall()                                         
                                                                                                         
          return jsonify([{                                                                              
              "vault_id": str(r.vault_id),                                                               
              "vault_name": r.vault_name,                                                                
              "document_id": str(r.document_id) if r.document_id else None,                              
              "document_name": r.document_name,                                                          
              "event_type": r.event_type,                                                                
              "extraction_level": r.extraction_level,                                                    
              "details": r.details,                                                                      
              "timestamp": r.created_at.isoformat(),                                                     
              "time_ago": format_time_ago(r.created_at)                                                  
          } for r in rows])                                                                              
                                                                                                         
                                                                                                         
  def format_time_ago(dt):                                                                               
      """Format timestamp as relative time."""                                                           
      from datetime import datetime, timezone                                                            
      now = datetime.now(timezone.utc)                                                                   
      diff = now - dt.replace(tzinfo=timezone.utc)                                                       
                                                                                                         
      if diff.seconds < 60:                                                                              
          return "just now"                                                                              
      elif diff.seconds < 3600:                                                                          
          return f"{diff.seconds // 60}m ago"                                                            
      elif diff.seconds < 86400:                                                                         
          return f"{diff.seconds // 3600}h ago"                                                          
      else:                                                                                              
          return f"{diff.days}d ago"

  Cancel Extraction:                                                                                     
  @app.route('/api/extraction/cancel/<vault_id>', methods=['POST'])                                      
  def cancel_extraction(vault_id):                                                                       
      """Cancel running extraction for a vault."""                                                       
      # Log the event                                                                                    
      log_extraction_event(vault_id, None, None, 'cancelled', details='Manually cancelled')              
                                                                                                         
      # Set a flag that the extraction loop checks                                                       
      with get_db_session() as session:                                                                  
          session.execute(text("""                                                                       
              INSERT INTO platform.extraction_events                                                     
              (vault_id, event_type, details)                                                            
              VALUES (:vault_id, 'cancel_requested', 'User requested cancellation')                      
          """), {"vault_id": vault_id})                                                                  
          session.commit()                                                                               
                                                                                                         
      return jsonify({"status": "cancel_requested", "vault_id": vault_id})

  def log_extraction_event(vault_id, vault_name, doc_id, doc_name, event_type, level='multi',            
  details=None):                                                                                         
      """Log an extraction event to the database."""                                                     
      with get_db_session() as session:                                                                  
          session.execute(text("""                                                                       
              INSERT INTO platform.extraction_events                                                     
              (vault_id, vault_name, document_id, document_name, event_type, extraction_level, details)  
              VALUES (:vault_id, :vault_name, :doc_id, :doc_name, :event_type, :level, :details)         
          """), {                                                                                        
              "vault_id": vault_id,                                                                      
              "vault_name": vault_name,                                                                  
              "doc_id": doc_id,                                                                          
              "doc_name": doc_name,                                                                      
              "event_type": event_type,                                                                  
              "level": level,                                                                            
              "details": details                                                                         
          })                                                                                             
          session.commit()

  function getDocumentStatusColor(doc) {                                                                 
      if (doc.status === 'failed') return 'red';                                                         
      if (doc.status === 'processing') return 'yellow';                                                  
      if (doc.extraction_level === 'multi') return 'blue';                                               
      if (doc.extraction_level === 'single') return 'green';                                             
      if (doc.status === 'extracted') return 'green';                                                    
      return 'gray';                                                                                     
  }                                                                                                      
                                                                                                         
  function getDocumentStatusLabel(doc) {                                                                 
      if (doc.status === 'failed') return 'Failed';                                                      
      if (doc.status === 'processing') return 'Processing';                                              
      if (doc.extraction_level === 'multi') return 'Multi-Model Done';                                   
      if (doc.extraction_level === 'single') return 'Single-Model Done';                                 
      if (doc.status === 'extracted') return 'Extracted';                                                
      return 'Pending';                                                                                  
  }
