TEST 2: FORCE A CANDIDATE TO BE CREATED

We need to test that unknown relationship types become candidates.

Option A: Create a test script
```python
# test_ontology_candidate.py

from src.context_foundry.ontology.candidate_store import CandidateStore
from src.context_foundry.ontology.normalizer import CandidateNormalizer
from src.context_foundry.models.schema import get_session
from uuid import UUID

# Use an existing tenant_id from your test data
TENANT_ID = UUID("your-tenant-id-here")  # Replace with actual tenant ID

def test_candidate_creation():
    session = get_session()
    store = CandidateStore(session, TENANT_ID)
    normalizer = CandidateNormalizer()
    
    # Test normalizer
    print("Testing normalizer...")
    assert normalizer.normalize_relationship("INVESTS_IN") == "INVESTED_IN"
    assert normalizer.normalize_relationship("BOUGHT") == "ACQUIRED"
    assert normalizer.normalize_relationship("COMPLETELY_NEW_TYPE") == "COMPLETELY_NEW_TYPE"
    print("✓ Normalizer working")
    
    # Create a test candidate
    print("\nCreating test candidate...")
    candidate_id = store.add_relationship_candidate(
        proposed_name="TEST_RELATIONSHIP",
        normalized_name="TEST_RELATIONSHIP",
        source_entity_type="ORGANIZATION",
        target_entity_type="ORGANIZATION",
        source_entity_name="TestCompanyA",
        target_entity_name="TestCompanyB",
        properties={"amount": "$1M"},
        original_text="TestCompanyA acquired TestCompanyB for $1M",
        document_id="test-doc-001"
    )
    print(f"✓ Candidate created: {candidate_id}")
    
    # Retrieve it
    print("\nRetrieving candidates...")
    candidates = store.get_pending_candidates()
    print(f"✓ Found {len(candidates)} candidates")
    
    # Find our test candidate
    test_candidate = next((c for c in candidates if c['normalized_name'] == 'TEST_RELATIONSHIP'), None)
    if test_candidate:
        print(f"✓ Test candidate found:")
        print(f"  - ID: {test_candidate['id']}")
        print(f"  - Confidence: {test_candidate['confidence_score']}")
        print(f"  - Documents: {test_candidate['document_count']}")
        print(f"  - Mentions: {test_candidate['mention_count']}")
    
    # Get pending extractions
    print("\nRetrieving pending extractions...")
    extractions = store.get_pending_extractions(candidate_id)
    print(f"✓ Found {len(extractions)} pending extractions")
    
    if extractions:
        print(f"  - Source: {extractions[0]['source_entity_name']}")
        print(f"  - Target: {extractions[0]['target_entity_name']}")
    
    # Get stats
    print("\nGetting stats...")
    stats = store.get_stats()
    print(f"✓ Stats: {stats}")
    
    # Add another mention (same type, different document)
    print("\nAdding second mention...")
    store.add_relationship_candidate(
        proposed_name="TEST_RELATIONSHIP",
        normalized_name="TEST_RELATIONSHIP",
        source_entity_type="ORGANIZATION",
        target_entity_type="ORGANIZATION",
        source_entity_name="TestCompanyC",
        target_entity_name="TestCompanyD",
        properties={"amount": "$2M"},
        original_text="TestCompanyC acquired TestCompanyD for $2M",
        document_id="test-doc-002"
    )
    
    # Check confidence increased
    candidates = store.get_pending_candidates()
    test_candidate = next((c for c in candidates if c['normalized_name'] == 'TEST_RELATIONSHIP'), None)
    if test_candidate:
        print(f"✓ Updated confidence: {test_candidate['confidence_score']}")
        print(f"  - Documents: {test_candidate['document_count']}")
        print(f"  - Mentions: {test_candidate['mention_count']}")
    
    print("\n✅ All tests passed!")

if __name__ == "__main__":
    test_candidate_creation()
```
