  -- CRITERION 1: Supply-chain edges (MOST CRITICAL)
  -- Need ≥3 of 5 docs to produce supply-chain edges

  -- A) How many docs produced supply-chain edges?
  SELECT
    COUNT(DISTINCT source_document_id) as docs_with_supply_chain_edges
  FROM cf_relationships
  WHERE tenant_id = '<vault-id>'
    AND relation_type IN ('SUPPLIES', 'CUSTOMER_OF', 'PROCURES_FROM',
  'VENDOR_OF', 'SUPPLIED_BY', 'SUPPLIER_OF');

  -- B) What are the actual supply-chain edges?
  SELECT
    r.source_document_id,
    r.relation_type,
    e1.canonical_name as source,
    e2.canonical_name as target
  FROM cf_relationships r
  JOIN cf_entities e1 ON r.source_entity_id = e1.id
  JOIN cf_entities e2 ON r.target_entity_id = e2.id
  WHERE r.tenant_id = '<vault-id>'
    AND relation_type IN ('SUPPLIES', 'CUSTOMER_OF', 'PROCURES_FROM',
  'VENDOR_OF', 'SUPPLIED_BY', 'SUPPLIER_OF')
  ORDER BY source_document_id, relation_type;
