2. Path Verification Script (Audit Automatically)
python# scripts/audit_query_paths.py
"""
Audit all query entry points to verify they use shared helpers.

Run this after any code changes to catch path divergence.
"""

import ast
import os
from pathlib import Path

REQUIRED_PATTERNS = {
    'web_app.py': {
        'must_call': ['ToolAgent.query', 'tool_agent.query'],
        'must_not_call': ['qa_verifier.verify', 'QAVerifier'],  # Should not run QA directly
        'must_not_contain': ['confidence =', 'qa_verdict ='],  # Should not calculate these
    },
    'tool_agent.py': {
        'must_call': ['build_qa_evidence', 'calculate_confidence', 'build_response'],
        'must_call_once': ['qa_verifier', 'QAVerifier'],  # QA should run exactly once
    },
}

def audit_file(filepath: str, rules: dict) -> list:
    """Audit a file against rules."""
    violations = []
    
    with open(filepath) as f:
        content = f.read()
    
    for pattern in rules.get('must_call', []):
        if pattern not in content:
            violations.append(f"MISSING: {filepath} must call {pattern}")
    
    for pattern in rules.get('must_not_call', []):
        if pattern in content:
            # Check if it's in the forbidden section (after ToolAgent.query returns)
            violations.append(f"VIOLATION: {filepath} should not call {pattern} directly")
    
    for pattern in rules.get('must_not_contain', []):
        # Check for direct assignment (not in shared helper)
        if pattern in content:
            violations.append(f"WARNING: {filepath} contains '{pattern}' - should use shared helper")
    
    return violations

def main():
    all_violations = []
    
    for filename, rules in REQUIRED_PATTERNS.items():
        filepath = find_file(filename)
        if filepath:
            violations = audit_file(filepath, rules)
            all_violations.extend(violations)
    
    if all_violations:
        print("❌ PATH AUDIT FAILED")
        for v in all_violations:
            print(f"  - {v}")
        return 1
    else:
        print("✅ PATH AUDIT PASSED")
        return 0

if __name__ == "__main__":
    exit(main())

### 4. Architectural Rule (Enforce Going Forward)
```
RULE: Single Query Entry Point

All query processing MUST go through ToolAgent.query().

Entry points (web_app.py, CLI, API) may ONLY:
1. Parse input
2. Call ToolAgent.query()
3. Format output

Entry points may NOT:
- Run QA verification
- Calculate confidence
- Build evidence
- Make LLM calls

These are INTERNAL to ToolAgent and use shared helpers.

Any PR that adds query logic outside ToolAgent.query() will be rejected.
```
