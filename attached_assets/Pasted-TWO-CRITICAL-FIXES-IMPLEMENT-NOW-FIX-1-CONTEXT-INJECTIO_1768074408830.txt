def verify_from_retrieval_result(self, question: str, answer: str, retrieval) -> QAVerdict:
    class RetrievalAdapter:
        def __init__(self, r):
            self.entities = r.entities if hasattr(r, 'entities') else []
            self.relationships = r.relationships if hasattr(r, 'relationships') else []
            self.chunks = r.chunks if hasattr(r, 'chunks') else []
            
            # Resolved role counts as entity evidence
            if hasattr(r, 'resolved_role') and r.resolved_role:
                self.entity_name = r.resolved_role.get('person_name')
                self.entity_found = True
            else:
                self.entity_found = len(self.entities) > 0
    
    return self.verify(question, answer, RetrievalAdapter(retrieval))
