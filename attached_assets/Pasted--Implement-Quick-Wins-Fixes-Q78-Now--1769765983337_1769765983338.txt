  def route_supplier_query(query: str):

                            'electrolyzer', 'battery', 'engine', 'sensor']

  def query_component_supplier(query: str, component: str):                                              
      """Traverse: Component <- SUPPLIES <- Supplier"""                                                  
      sql = """                                                                                          
          SELECT s.name as supplier, c.name as component, p.name as program                              
          FROM entities s                                                                                
          JOIN relationships r1 ON s.id = r1.source_id                                                   
          JOIN entities c ON r1.target_id = c.id                                                         
          JOIN relationships r2 ON c.id = r2.source_id                                                   
          JOIN entities p ON r2.target_id = p.id                                                         
          WHERE r1.relationship_type = 'SUPPLIES'                                                        
          AND r2.relationship_type = 'COMPONENT_OF'                                                      
          AND c.name ILIKE :component                                                                    
          AND s.tenant_id = :tenant_id                                                                   
      """                                                                                                
      return execute(sql, {'component': f'%{component}%', 'tenant_id': tenant_id})
