```python
def get_tenant_id_by_name(vault_name: str) -> str:
    """Look up tenant ID by vault name at runtime."""
    result = session.execute(
        text("""
            SELECT id FROM platform.tenants 
            WHERE name ILIKE :name 
            LIMIT 1
        """),
        {"name": f"%{vault_name}%"}
    ).fetchone()
    
    if not result:
        raise ValueError(f"Vault '{vault_name}' not found in platform.tenants")
    
    return str(result.id)
```
