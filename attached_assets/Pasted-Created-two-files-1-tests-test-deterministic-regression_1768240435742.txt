  def test_known_types_not_in_candidates(self, db_session):
      """Known types should NEVER appear as pending candidates."""
      result = db_session.execute(text("""
          SELECT normalized_name FROM ontology_candidates
          WHERE normalized_name = ANY(:known_types)
            AND status = 'PENDING'
      """), {"known_types": KNOWN_TYPES_MUST_NOT_BE_CANDIDATES})

      violations = list(result)
      assert len(violations) == 0, f"REGRESSION: {violations}"
