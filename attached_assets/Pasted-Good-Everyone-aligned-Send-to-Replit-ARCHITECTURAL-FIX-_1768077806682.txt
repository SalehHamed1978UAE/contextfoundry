```python
class DocumentSearcher:
    """Single source of truth for document search."""
    
    def __init__(self, session, tenant_id):
        self.session = session
        self.tenant_id = tenant_id
    
    def search(self, query: str, limit: int = 5, offset: int = 0) -> List[dict]:
        """
        Search documents using consistent logic.
        
        1. Try vector similarity search
        2. Fall back to text search with LIKE
        3. Return chunks with metadata
        """
        # Vector search first
        chunks = self._vector_search(query, limit + offset)
        
        # If insufficient, supplement with text search
        if len(chunks) < limit:
            text_chunks = self._text_search(query, limit, offset)
            # Merge and dedupe
            chunks = self._merge_results(chunks, text_chunks, limit)
        
        return chunks[offset:offset + limit]
    
    def _text_search(self, query: str, limit: int, offset: int = 0) -> List[dict]:
        """
        Text search using LIKE with OR logic for keywords.
        This is the same logic that works in RetrievalRouter.
        """
        keywords = [w.strip('?.,!') for w in query.split() if len(w.strip('?.,!')) >= 3]
        
        if not keywords:
            return []
        
        # OR logic for keywords (any keyword matches)
        conditions = " OR ".join([f"LOWER(dc.text) LIKE :kw{i}" for i in range(len(keywords))])
        params = {f"kw{i}": f"%{kw.lower()}%" for i, kw in enumerate(keywords)}
        params["tenant_id"] = self.tenant_id
        params["limit"] = limit
        params["offset"] = offset
        
        sql = f"""
            SELECT dc.id, dc.text, d.title as document_name
            FROM document_chunks dc
            JOIN documents d ON dc.document_id = d.id
            WHERE dc.tenant_id = :tenant_id
            AND ({conditions})
            ORDER BY dc.created_at DESC
            LIMIT :limit OFFSET :offset
        """
        
        result = self.session.execute(text(sql), params)
        return [{"id": row.id, "text": row.text, "document_name": row.document_name} for row in result.fetchall()]
```

- ToolExecutor to use DocumentSearcher

```python
def _can_answer_directly(self, query: str, retrieval: RetrievalResult) -> bool:
    """
    Determine if pre-retrieval gathered enough to answer without tool loop.
    
    This is the KEY DECISION POINT:
    - True → Synthesize directly from pre-fetched data (skip tools)
    - False → Run ReAct tool loop
    """
    classification = retrieval.classification
    
    # If we resolved a role and found chunks, we have the answer
    if retrieval.resolved_role and len(retrieval.chunks) >= 3:
        logger.info("[AGENT] Direct answer: role resolved + chunks")
        return True
    
    # Relationship queries with relationships found
    if classification.query_type == "RELATIONSHIP" and len(retrieval.relationships) >= 1:
        logger.info("[AGENT] Direct answer: relationship query + relationships found")
        return True
    
    # Attribute queries with sufficient chunks
    if classification.query_type == "ATTRIBUTE" and len(retrieval.chunks) >= 2:
        logger.info("[AGENT] Direct answer: attribute query + chunks")
        return True
    
    # EXPLORATION queries - THIS WAS MISSING
    if classification.query_type == "EXPLORATION":
        has_sufficient_chunks = len(retrieval.chunks) >= 3
        has_entity_context = len(retrieval.entities) >= 1 and len(retrieval.relationships) >= 2
        
        if has_sufficient_chunks:
            logger.info(f"[AGENT] Direct answer: exploration + {len(retrieval.chunks)} chunks")
            return True
        
        if has_entity_context:
            logger.info(f"[AGENT] Direct answer: exploration + entity with relationships")
            return True
    
    # AGGREGATION queries with list data
    if classification.query_type == "AGGREGATION" and classification.expects_list:
        if len(retrieval.chunks) >= 5 or len(retrieval.relationships) >= 3:
            logger.info("[AGENT] Direct answer: aggregation with list data")
            return True
    
    logger.info(f"[AGENT] Insufficient for direct answer: type={classification.query_type}, "
                f"chunks={len(retrieval.chunks)}, entities={len(retrieval.entities)}, "
                f"rels={len(retrieval.relationships)}")
    return False
```
