  # Extract all failing questions                                                                        
  python3 -c "                                                                                           
  import json                                                                                            
  import glob                                                                                            
                                                                                                         
  for f in glob.glob('test_results/*progress*.jsonl'):                                                   
      with open(f) as file:                                                                              
          for line in file:                                                                              
              if line.strip():                                                                           
                  d = json.loads(line)                                                                   
                  if not d.get('passed'):                                                                
                      print(f\"Q{d['q']}: {d.get('query', '')[:60]}\")                                   
                      print(f\"  Expected: {d.get('expected', '')[:60]}\")                               
                      print(f\"  Got: {d.get('actual', '')[:80]}\")                                      
                      print(f\"  Failure: {d.get('failure_category', 'unknown')}\")                      
                      print()                                                                            
  "
