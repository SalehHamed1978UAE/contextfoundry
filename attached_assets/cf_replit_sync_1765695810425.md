-- 3. Check overall duplicate exposure
SELECT name, entity_type, COUNT(*) as dupes
FROM entities 
WHERE lifecycle_state = 'TRUSTED'
GROUP BY name, entity_type
HAVING COUNT(*) > 1
ORDER BY dupes DESC
LIMIT 20;

-- 4. RE Agent progress check
SELECT 
  COUNT(*) FILTER (WHERE lifecycle_state = 'TRUSTED') as trusted,
  COUNT(*) FILTER (WHERE lifecycle_state = 'STAGING') as staging,
  COUNT(*) as total
FROM relationships;
