### Phase 2: Re-extract all documents

```python
import sys
sys.path.insert(0, '/home/runner/workspace')

from uuid import UUID
from src.context_foundry.models.schema import get_session, Document
from src.context_foundry.extraction.extraction_pipeline import ExtractionPipeline
from src.context_foundry.ingestion.ingestion_pipeline import IngestionPipeline

tenant_id = "f3dd3201-7225-4a55-9264-445f99d0eba3"

with get_session() as session:
    documents = session.query(Document).filter(
        Document.tenant_id == UUID(tenant_id)
    ).all()
    
    print(f"Re-extracting {len(documents)} documents...")
    
    for i, doc in enumerate(documents):
        try:
            # Create IngestedDocument from existing document
            ingested = IngestionPipeline.from_existing_document(doc)
            
            # Extract entities and relations
            pipeline = ExtractionPipeline(session, tenant_id=tenant_id)
            result = pipeline.extract_from_document(ingested)
            
            session.commit()
            print(f"  [{i+1}/{len(documents)}] {doc.filename}: {result.entities_created} entities")
            
        except Exception as e:
            print(f"  [{i+1}/{len(documents)}] {doc.filename}: ERROR - {e}")
            session.rollback()
    
    print("Done")
```
