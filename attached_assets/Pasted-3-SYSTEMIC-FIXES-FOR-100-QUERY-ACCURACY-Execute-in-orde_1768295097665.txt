FIX 1: PORTFOLIO COMPANY PATTERN (30 min)
================================================================================

ADD TO post_processor.py:

PORTFOLIO_COMPANY_PATTERNS = [
    # "CLOUDAI (Cohort 8)" or "CloudAI (Cohort 8)"
    (r'^([A-Z][A-Za-z0-9]+)\s*\(Cohort\s+\d+\)', 'PORTFOLIO_COMPANY'),
    
    # "COMPANYNAME\n- Sector:" format  
    (r'^([A-Z][A-Za-z0-9]+)\s*\n\s*-\s*Sector:', 'PORTFOLIO_COMPANY'),
]

def _extract_portfolio_companies(self, text: str) -> List[dict]:
    """Extract portfolio companies using patterns"""
    entities = []
    seen = set()
    
    for pattern, entity_type in PORTFOLIO_COMPANY_PATTERNS:
        for match in re.finditer(pattern, text, re.MULTILINE):
            name = match.group(1).strip()
            if name not in seen and len(name) >= 3:
                seen.add(name)
                entities.append({
                    'name': name,
                    'entity_type': entity_type,
                    'confidence': 0.9,
                    'source_text': match.group(0)[:100]
                })
    
    return entities
```

import logging
logger = logging.getLogger(__name__)

def query_role(role_name, vault_id):
    """Query for a person by role"""
    
    expanded = expand_role_acronym(role_name)
    
    # Step 1: Try exact relationship match
    results = db.query("""
        SELECT person.name, role.name as role_name
        FROM entities person
        JOIN relationships r ON person.id = r.source_entity_id
        JOIN entities role ON r.target_entity_id = role.id
        WHERE r.relationship_type IN ('HOLDS_POSITION', 'HAS_ROLE')
          AND r.tenant_id = %s
          AND (role.name ILIKE %s OR role.name ILIKE %s)
    """, [vault_id, f'%{role_name}%', f'%{expanded}%'])
    
    if results:
        logger.info(f"Role query '{role_name}' found {len(results)} matches via relationship")
        return {
            "answer": format_role_results(results),
            "confidence": 0.9
        }
    
    # Step 2: NO SEMANTIC FALLBACK
    logger.info(f"Role query '{role_name}' no relationship match - returning not found")
    return {
        "answer": f"No {role_name} found in the knowledge graph.",
        "confidence": 0.0
    }

def expand_role_acronym(role: str) -> str:
    expansions = {
        'CTO': 'Chief Technology Officer',
        'CFO': 'Chief Financial Officer',
        'CEO': 'Chief Executive Officer',
        'COO': 'Chief Operating Officer',
        'CMO': 'Chief Marketing Officer',
        'CIO': 'Chief Information Officer',
    }
    return expansions.get(role.upper(), role)
```
